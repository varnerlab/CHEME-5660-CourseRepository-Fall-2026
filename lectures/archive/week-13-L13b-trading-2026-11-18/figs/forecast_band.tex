% Drawing commands for Fig-L13b-ForecastBand.
% \fbband{<file>}{<tag>}{<first day>}{<last day>}: the 95% band (mean +/- 1.96 sd) of one plan,
% filled, with both edges drawn as thin lines, and its mean as a dashed line.
\newcommand{\fbband}[4]{%
  \addplot[name path=fb#2hi, draw=vnband, line width=0.6pt, forget plot,
    restrict x to domain=#3:#4]
    table[col sep=comma, x=day, y expr={\thisrow{mean} + 1.96*\thisrow{sd}}] {#1};
  \addplot[name path=fb#2lo, draw=vnband, line width=0.6pt, forget plot,
    restrict x to domain=#3:#4]
    table[col sep=comma, x=day, y expr={\thisrow{mean} - 1.96*\thisrow{sd}}] {#1};
  \addplot[fill=vnband, fill opacity=\vnbandinner, draw=none, forget plot]
    fill between[of=fb#2hi and fb#2lo];
  \addplot[draw=vnslidelink, line width=0.8pt, dash pattern=on 3pt off 2pt, forget plot,
    restrict x to domain=#3:#4]
    table[col sep=comma, x=day, y=mean] {#1};
}
