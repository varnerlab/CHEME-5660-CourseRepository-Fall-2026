\begin{frame}{Monte Carlo Pricing of American Options}
  We simulate share price paths under $\Q$ and, like the lattice, work backward
  from expiration. At each date, a regression on the share price estimates the
  value of holding.

  \vspace{0.6em}
  \hilite{Initialize}: given the share price $S_{0}$, strike price $K$, payoff
  $V(K,S)$ of the call or put, time to expiration $T$ (years), volatility
  $\sigma$, risk-free rate $g_{y}$, number of exercise dates $h$ after today,
  and number of paths $M$. Compute $\Delta t=T/h$ and the one-step discount
  factor $\Dstep=e^{-g_{y}\Delta t}$.
  \slidenote{Reference:}{Longstaff, F. A., and Schwartz, E. S. (2001). Valuing American options by simulation: a simple least-squares approach. \href{https://doi.org/10.1093/rfs/14.1.113}{Review of Financial Studies 14(1): 113–147}.}
\end{frame}

\begin{frame}{Pseudocode: Longstaff–Schwartz}
  \begin{enumerate}\setlength{\itemsep}{0.3em}
    \item \hilite{Simulate} $M$ paths under $\Q$ from $S_{0}$ on the dates
      $t=0,1,\dots,h$; $S_{i}(t)$ is the share price on path $i$ at date $t$.
    \item At expiration, each path's cash flow is its payoff:
      $Y_{i}(h)\gets V(K,S_{i}(h))$ for $i=1,2,\dots,M$, where $Y_{i}(t)$ is
      path $i$'s cash flow valued at date $t$.
    \item For each date $t=h-1,h-2,\dots,1$ \textbf{do}:
      \begin{itemize}
        \item[a.] Discount one date: $Y_{i}(t)\gets\Dstep\cdot Y_{i}(t+1)$.
        \item[b.] \hilite{Regress}: on the in-the-money paths ($V(K,S_{i}(t))>0$), fit
          $\hat{H}_{t}(S)=\beta_{0}+\beta_{1}S+\beta_{2}S^{2}$ to the pairs
          $\bigl(S_{i}(t),Y_{i}(t)\bigr)$ by least squares.
        \item[c.] \hilite{Decide}: on each in-the-money path, if
          $V(K,S_{i}(t))>\hat{H}_{t}(S_{i}(t))$, \textbf{then} exercise:
          $Y_{i}(t)\gets V(K,S_{i}(t))$.
      \end{itemize}
    \item Discount to today and average:
      $\bar{Y}\gets\mathcal{D}^{-1}_{1,0}(g_{y})\cdot\frac{1}{M}\sum_{i=1}^{M}Y_{i}(1)$.
    \item The premium is $\PAM\gets\max\bigl\{V(K,S_{0}),~\bar{Y}\bigr\}$.
  \end{enumerate}
\end{frame}

\begin{frame}{Intrinsic and Extrinsic Value}
  The premium of an American option has two parts:
  \begin{itemize}\setlength{\itemsep}{0.4em}
    \item \hilite{Intrinsic value}: what exercising today pays (red dashed
      line), $\max(0, S(t) - K)$ for a call and $\max(0, K - S(t))$ for a put,
      with share price $S(t)$ and strike price $K$.
    \item \hilite{Extrinsic value} (time value): the rest of the premium, the
      gap above the red dashed line.
  \end{itemize}
  \begin{equation*}
    \boxed{
    \begin{aligned}
      E_{V}(t) = \text{Option Price} - \text{Intrinsic Value}\quad\blacksquare
    \end{aligned}}
  \end{equation*}
\end{frame}

\begin{frame}{Extrinsic Value}
  With call and put premiums $\mathcal{P}_{c}$ and $\mathcal{P}_{p}$, the
  extrinsic value is:
  \begin{equation*}
    \boxed{
    \begin{aligned}
      E_{V}(t) &= \begin{cases}
      \mathcal{P}_{c}(K,S(t)) - \max(0, S(t) - K) & \text{(for Call Options)} \\
      \mathcal{P}_{p}(K,S(t)) - \max(0, K - S(t)) & \text{(for Put Options)}
      \end{cases}
    \end{aligned}}
  \end{equation*}
  \begin{itemize}\setlength{\itemsep}{0.4em}
    \item \hilite{At or out of the money} (points a, c): no intrinsic value, so
      the whole premium is extrinsic, largest at a.
    \item \hilite{Time decay}: extrinsic value shrinks as expiration nears.
    \item \hilite{Why pay for ``worthless''?} The share price can still move
      favorably before expiration.
  \end{itemize}
\end{frame}

\begin{frame}{Would You Ever Exercise Early?}
  \begin{itemize}\setlength{\itemsep}{0.15em}
    \item \hilite{A call}: no, not without dividends. Exercising buys the shares
      at $K$ today. Holding, we pay $K$ later (earning interest on it) and skip
      buying if the share price falls below $K$.
    \item \hilite{A put}: yes, far in the money. Exercising sells the shares at
      $K$ today, and that cash earns interest.
  \end{itemize}

  \end{center}
\end{frame}

\begin{frame}{The Greeks}
  The \hilite{Greeks} measure how the premium changes with market conditions:
  \begin{itemize}\setlength{\itemsep}{0.35em}
    \item \hilite{Delta $\Delta$}: premium change per \texttt{+1~USD/share} in
      the share price.
    \item \hilite{Theta $\Theta$}: premium change per \texttt{1-day decrease}
      in the contract's lifetime (time decay), quoted as $\Theta_{\star}/365$.
    \item \hilite{Vega $\nu$}: premium change per \texttt{+1 point} in volatility,
      quoted as $\nu_{\star}/100$.
    \item \hilite{Gamma $\Gamma$}: change in $\Delta$ per \texttt{+1~USD/share}
      in the share price.
    \item \hilite{Rho $\rho$}: premium change per \texttt{+1 point} in the
      risk-free rate, quoted as $\rho_{\star}/100$; minor except for long-dated options.
  \end{itemize}
  Trading platforms compute these using the implied volatility under $\Q$.
\end{frame}

\begin{frame}{A Local Taylor Expansion of the Premium}
  Expanding the premium $\Pst(S,\sigma,t,g_{y})$ of a contract of type
  $\star\in\{c,p\}$ at calendar time $t$ to second order in $S$ and first
  order in the other inputs gives:
  \begin{equation*}
    \boxed{
    \begin{aligned}
      \Delta\Pst &\approx \Delta_{\star}\,\Delta S + \nu_{\star}\,\Delta\sigma
      + \Theta_{\star}\,\Delta t + \rho_{\star}\,\Delta g_{y}
      + \tfrac{1}{2}\,\Gamma_{\star}\,(\Delta S)^{2}\\
      &= \underbrace{\bigl(\Delta_{\star} + \tfrac{1}{2}\,\Gamma_{\star}\,\Delta S\bigr)\Delta S}_{\text{share price}}
      + \underbrace{\nu_{\star}\,\Delta\sigma}_{\text{volatility}}
      + \underbrace{\Theta_{\star}\,\Delta t}_{\text{time}}
      + \underbrace{\rho_{\star}\,\Delta g_{y}}_{\text{rate}}
    \end{aligned}}
  \end{equation*}
  \begin{equation*}
    \Delta_{\star}=\frac{\partial\Pst}{\partial S}\qquad
    \nu_{\star}=\frac{\partial\Pst}{\partial \sigma}\qquad
    \Theta_{\star}=\frac{\partial\Pst}{\partial t}\qquad
    \rho_{\star}=\frac{\partial\Pst}{\partial g_{y}}\qquad
    \Gamma_{\star}=\frac{\partial^{2}\Pst}{\partial S^{2}}
  \end{equation*}
  One day changes the premium by $\Theta_{\star}/365$; a one-point rise in the
  volatility or the rate changes it by $\nu_{\star}/100$ or $\rho_{\star}/100$.
\end{frame}

\begin{frame}{Delta and Gamma}
  Delta $\Delta_{\star}$ and Gamma $\Gamma_{\star}$ are the slope and curvature
  of the premium against the share price.

  \vspace{0.5em}
  \textbf{\ink{Example:}} a long call with $\Delta_{c}=0.50$ and
  $\Gamma_{c}=0.02$ per USD/share:
  \begin{align*}
    \Delta S=+5:\quad \Delta\mathcal{P}_{c} &\approx \bigl(0.50+\tfrac{1}{2}(0.02)(5)\bigr)\times5=0.55\times5=2.75~\text{USD/share}\\
    \Delta S=-5:\quad \Delta\mathcal{P}_{c} &\approx (0.50-0.05)\times(-5)=-2.25~\text{USD/share}
  \end{align*}
  Gamma adds $\tfrac{1}{2}\Gamma_{c}(\Delta S)^{2}=0.25$~USD/share to the
  Delta-only $\pm2.50$: the call gains more going up and loses less going down.
\end{frame}

\begin{frame}{Computing the Greeks of American Options}
  The tree gives no formula to differentiate, so we reprice it with one input
  moved up and down by $\epsilon$, a
  \href{https://en.wikipedia.org/wiki/Finite_difference\#Basic_types}{\hilite{central finite difference}}:
  \begin{equation*}
    \boxed{
    \begin{aligned}
      \frac{\partial\Pst}{\partial x} & \approx \frac{\Pst(x+\epsilon)-\Pst(x-\epsilon)}{2\epsilon}\qquad x\in\{S,\sigma,t,g_{y}\}\\
      \Gamma_{\star} & \approx \frac{\Pst(S+\epsilon)-2\,\Pst(S)+\Pst(S-\epsilon)}{\epsilon^{2}}
    \end{aligned}}
  \end{equation*}
  With enough tree steps, the CRR tree estimates Delta well, but its Gamma can be
  zero or erratic for a small step $\epsilon$, so we use the \hilite{Leisen--Reimer tree}, a modified CRR tree.
  \slidenote{Example 2:}{\href{\repo/CHEME-5660-L10a-Example-Greeks-PutCall-Fall-2026.ipynb}{\dimmed{CHEME-5660-L10a-Example-Greeks-PutCall-Fall-2026.ipynb}}}
\end{frame}

\begin{frame}{The Secret Lives of Delta}
  Besides the premium change per \texttt{+1~USD/share}, $\Delta$ has two other
  meanings:
  \begin{itemize}\setlength{\itemsep}{0.6em}
    \item \hilite{Share equivalent}: $n$ contracts ($n<0$ if sold) gain or lose
      as much as $100\cdot n\cdot\Delta$ shares for small moves. Holding
      $-100\cdot n\cdot\Delta$ shares makes the position
      \hilite{delta-neutral}, so Delta is the \hilite{hedge ratio}.
    \item \hilite{Probability of exercise}: Delta estimates the probability of
      expiring in the money; a call with $\Delta=0.7$ suggests about 70\% under $\Q$.
  \end{itemize}
\end{frame}

\begin{frame}{Probability of Exercise}
  \vspace{0.3em}
  \hilite{Probability of exercise (POE)}: the probability that the contract
  finishes in the money at expiration $t=T$:
  \begin{equation*}
    \boxed{
    \begin{aligned}
      \text{POE} & = \begin{cases}
      \mathbb{P}\left(S_{T} > K + 0.01\right) & \text{(for Call Options)} \\
      \mathbb{P}\left(S_{T} < K - 0.01\right) & \text{(for Put Options)}
      \end{cases}
    \end{aligned}}
  \end{equation*}
  where $S_{T}$ is the share price at expiration and $K$ is the strike price.\par
  \medskip
  \textbf{\ink{Why 0.01~USD/share?}} OCC automatically exercises equity options
  \href{https://www.optionseducation.org/referencelibrary/faq/options-exercise}{\hilite{in the money by 0.01~USD/share or more}}.
\end{frame}

\begin{frame}{Why Is Delta Close to the POE?}
  Quoted Deltas come from pricing models under $\Q$, so we compare Delta with
  the POE under $\Q$. \textbf{\ink{Proposition:}} for a call (no dividends) on an $h$-step
  CRR tree under $\Q$, both sum over the same in-the-money nodes,
  $\text{ITM}=\left\{k : S_{0}u^{k}d^{h-k}>K\right\}$ with $k$ up moves:
  \begin{equation*}
    \boxed{
    \begin{aligned}
    \Delta_{c} & = \sum_{k\,\in\,\text{ITM}}\binom{h}{k}\left(q^{*}\right)^{k}\left(1-q^{*}\right)^{h-k} && \text{quoted Delta: up probability } q^{*}\\
    \Q\left(S_{T}>K\right) & = \sum_{k\,\in\,\text{ITM}}\binom{h}{k}\,q^{k}\left(1-q\right)^{h-k} && \text{POE under } \Q\text{: up probability } q
    \end{aligned}}
  \end{equation*}
  Here $u$, $d$, $q$, and $\Delta t$ are the tree's up and down factors, up
  probability, and step, and $q^{*}=q\,u\,e^{-g_{y}\Delta t}$ is $q$ times $u$ over the
  risk-free growth $e^{g_{y}\Delta t}$ in one step. \hilite{Delta is larger than the POE
  under $\Q$}: $u>e^{g_{y}\Delta t}$, so $q^{*}>q$.
  \slidenote{Derivation:}{\href{\repo/advanced/delta_poe/CHEME-5660-L10a-Derivation-Delta-POE-Fall-2026.ipynb}{\dimmed{advanced/delta\_poe/CHEME-5660-L10a-Derivation-Delta-POE-Fall-2026.ipynb}}}
\end{frame}

\begin{frame}{When Is Delta Close to the POE?}
  \begin{itemize}\setlength{\itemsep}{0.6em}
    \item \hilite{Volatility and time to expiration}: the gap grows with both,
      so Delta is close to the POE under $\Q$ for short-dated contracts.
    \item \hilite{Our forecast and American puts}: our forecast POE under
      $\mathbb{P}$ puts the real-world up probability $p$ in place of $q$. Nothing
      ties $p$ to $q^{*}$, so it can sit above or below Delta: Delta follows
      $\Q$, not our belief. Early exercise adds to an American put's $|\Delta_{p}|$, so the
      match is approximate.
  \end{itemize}
\end{frame}

\begin{frame}{Probability of Profit}
  \vspace{0.3em}
  \hilite{Probability of profit (POP)}: the probability that the profit at
  expiration $\hat{P}_{\mathcal{C}}$ is positive:
  \begin{equation*}
    \boxed{\text{POP}=\mathbb{P}\left(\hat{P}_{\mathcal{C}} > 0\right)}
  \end{equation*}
  For a long call, the profit is the payoff minus the premium's future value:
  \begin{align*}
    \hat{P}_{\mathcal{C}} > 0 & \iff \left(S_{T}-K\right)^{+} - \mathcal{D}_{T,0}(g_{y})\cdot\mathcal{P}_{c} > 0 && \text{profit of a long call}\\
    & \iff \left(S_{T}-K\right)^{+} > \mathcal{D}_{T,0}(g_{y})\cdot\mathcal{P}_{c} && \text{add the premium's future value}\\
    & \iff S_{T}-K > \mathcal{D}_{T,0}(g_{y})\cdot\mathcal{P}_{c} && \text{a positive payoff needs } S_{T}>K\\
    & \iff S_{T} > \underbrace{K + \mathcal{D}_{T,0}(g_{y})\cdot\mathcal{P}_{c}}_{=\,S^{\ddagger}} && \text{the breakeven share price}
  \end{align*}
\end{frame}

\begin{frame}{Probability of Profit and the Breakeven}
  For a put, $S_{T}<S^{\ddagger}$ with
  $S^{\ddagger}=K-\mathcal{D}_{T,0}(g_{y})\cdot\mathcal{P}_{p}$. The seller
  profits exactly when the buyer does not:
  \begin{equation*}
    \boxed{
    \text{POP} = \begin{cases}
    \mathbb{P}\left(S_{T} > S^{\ddagger}\right) & \text{(bought a call or sold a put)} \\
    \mathbb{P}\left(S_{T} < S^{\ddagger}\right) & \text{(sold a call or bought a put)}
    \end{cases}}
  \end{equation*}
  The POP is the POE with the breakeven in place of the strike, so the buyer's
  POP is smaller than the POE. We compute both under our belief $\mathbb{P}$:
  GBM with our estimates of $\mu$ and $\sigma$.
\end{frame}

\begin{frame}{Computing POE and POP by Simulation}
  \vspace{0.3em}
  \hilite{Initialize}: given $S_{0}$, $K$, the premium $\mathcal{P}$ of the
  call or put, $T$ (years), the drift $\mu$ and volatility $\sigma$ under
  $\mathbb{P}$, $g_{y}$, and the number of samples $M$. The breakeven is
  $S^{\ddagger}=K+\mathcal{D}_{T,0}(g_{y})\cdot\mathcal{P}$ for a call or
  $S^{\ddagger}=K-\mathcal{D}_{T,0}(g_{y})\cdot\mathcal{P}$ for a put, with
  $\mathcal{D}_{T,0}(g_{y})=e^{g_{y}T}$.
  \begin{enumerate}\setlength{\itemsep}{0.35em}
    \item For each sample $i=1,2,\dots,M$ \textbf{do}:
      \begin{itemize}
        \item[a.] Draw $Z_{i}\sim\mathcal{N}(0,1)$.
        \item[b.] Compute
          $S_{i}(T)\gets S_{0}\exp\left[\left(\mu-\sigma^{2}/2\right)T+\sigma\sqrt{T}\,Z_{i}\right]$.
      \end{itemize}
    \item Compute the fraction of samples at or below each share price $x$:
      $\hat{F}(x)\gets\frac{1}{M}\cdot\#\left\{i:S_{i}(T)\leq x\right\}$.
  \end{enumerate}
\end{frame}

\begin{frame}{Pseudocode: POE and POP}
  \vspace{0.3em}
  \begin{enumerate}\setlength{\itemsep}{0.45em}
    \setcounter{enumi}{2}
    \item If the contract is a call, \textbf{then}
      $\text{POE}\gets1-\hat{F}(K+0.01)$. Otherwise,
      $\text{POE}\gets\hat{F}(K-0.01)$.
    \item Compute the probability of profit:
      \begin{itemize}\setlength{\itemsep}{0.25em}
        \item If we bought a call or sold a put, \textbf{then}
          $\text{POP}\gets1-\hat{F}(S^{\ddagger})$.
        \item If we sold a call or bought a put, \textbf{then}
          $\text{POP}\gets\hat{F}(S^{\ddagger})$.
      \end{itemize}
  \end{enumerate}

  \vspace{0.6em}
  With $\mu=g_{y}$ and $\sigma$ set to the implied volatility, the same steps
  give the probabilities under $\Q$. These are the probabilities inside the
  market price, not our forecast.

\end{frame}

\begin{frame}{Summary}
  \vspace{0.3em}
  {\large\textbf{\ink{Key takeaways}}}
  \begin{itemize}\setlength{\itemsep}{0.5em}
    \item \hilite{Estimating and decomposing the American premium}: we
      regressed the value of holding on the share price along simulated paths.
      Extrinsic value is largest at the money; without dividends, only a deep
      in-the-money put is worth exercising early.
    \item \hilite{The Greeks quantify option price sensitivity}: we estimated
      them by finite differences on a binomial tree, as coefficients of a local
      Taylor expansion of the premium.
    \item \hilite{Delta serves three purposes}: premium change per USD/share,
      share equivalent, and an estimate of a call's POE. The POP is the POE
      with the breakeven in place of the strike.
  \end{itemize}
\end{frame}
