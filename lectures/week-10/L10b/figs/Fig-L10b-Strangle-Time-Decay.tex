% L10b short strangle before expiration, two panels. Why the composite-contract
% diagrams are drawn at expiration.
%
% A short strangle is opened 30 days before expiration (sell a put at K1 = 90 and a
% call at K2 = 110 USD/share with S(0) = 100). If we buy both legs back early, the
% profit is the credit, grown to that day, minus what the legs cost then; each leg
% still carries extrinsic (time) value, so the curve sits below the profit at
% expiration and rises onto it as the time value decays.
% Left: the profit against the share price on the day we close, with 30 (entry),
% 20, and 10 days left (orange, darker closer to expiration) and at expiration (ink).
% Right: if the share price is still S(0), the credit collected at entry (dashed,
% grown with interest) and the cost to buy both legs back (orange), against the days
% to expiration. The gap is the profit if we close: zero at entry, where the cost
% equals the credit, and the whole credit at expiration, where both legs expire
% worthless. The four days are marked in the left panel's colors.
%
% The curves are computed, not sketched: generate-strangle-time-decay.jl prices the
% American legs on CRR trees and writes Fig-L10b-Strangle-Time-Decay-coords.tex.
% Left panel x = 0.28125 (S - 84), so S = 84 ... 116 USD/share spans 0 ... 9.0;
% left panel 0.6 units per USD/share. Right panel x = 0.5 + 0.273 (days since
% entry) and 1.9 units per USD/share above a zero line at 0.55.
\documentclass[tikz,border=4pt]{standalone}
\input{profit-diagram.tex}
\input{Fig-L10b-Strangle-Time-Decay-coords.tex}
\def\yscale{0.6}
\def\yzero{4.3}

\tikzset{
  early/.style={line width=1.2pt, line join=round},
  day30/.style={early, draw=saorange!50},
  day20/.style={early, draw=saorange!70},
  day10/.style={early, draw=saorange},
  key/.style={anchor=west, font=\sffamily\footnotesize, inner sep=2pt},
  credit/.style={draw=vnink, line width=1.2pt, dash pattern=on 5pt off 3pt},
  cost/.style={draw=saorange, line width=1.6pt, line join=round},
}

% A day on the right panel's axis: a short tick at day #1 with the days to
% expiration #2 and, below it, #3.
\newcommand{\daytick}[3]{%
  \draw[draw=vnink, line width=0.8pt] (#1, -\yzero/\yscale) -- (#1, -\yzero/\yscale + 0.12/\yscale);
  \node[anchor=north, align=center, inner sep=1.5pt, text=gdgray]
    at (#1, -\yzero/\yscale - 0.08/\yscale) {#2\\[-1.5pt]{\footnotesize\color{vnink}#3}};}

% Scope in which (share price, profit) are the coordinates.
\newenvironment{sharedata}{\begin{scope}[shift={(-23.625, \yzero)}, xscale=0.28125, yscale=\yscale]}{\end{scope}}
% Scope in which (days since entry, profit) are the coordinates.
\newenvironment{daydata}{\begin{scope}[shift={(0.5, \yzero)}, xscale=0.273, yscale=\yscale]}{\end{scope}}

      \draw[draw=vnink, line width=0.8pt] (100, -\yzero/\yscale) -- (100, -\yzero/\yscale + 0.12/\yscale);
