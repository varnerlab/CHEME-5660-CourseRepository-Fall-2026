\begin{frame}{The Idea}
  \vspace{0.3em}
  An investor with Cobb-Douglas utility splits the wealth in proportion to the preference weights:
  \begin{itemize}\setlength{\itemsep}{0.5em}
    \item \hilite{Long only}: the rule never sells a ticker short.
    \item \hilite{Signal}: a ticker with a higher preference weight gets more of the wealth, and a ticker with a preference weight that is not positive gets nothing.
    \item \hilite{Default}: the examples start with Cobb-Douglas, as the practicum does.
  \end{itemize}

  \vspace{0.5em}
  The basket comes from the picking rule. Each preference weight is the day's value at the decision.
\end{frame}

\begin{frame}{The Preference Weight}
  \vspace{0.3em}
  At each close $t$, the engine scores every ticker $i$ with its preference weight $\gamma_{i}(t)$:
  {\setlength{\jot}{3pt}
  \begin{align*}
    \tilde{g}_{i,t} &= \alpha_{i}(t)+\beta_{i}(t)\,\tilde{g}_{M,t} && \text{the expected growth rate (1/yr)}\\
    \gamma_{i}(t) &= \tanh\left(h\,\frac{\tilde{g}_{i,t}}{\max\left(|\beta_{i}(t)|,\,\beta_{\text{floor}}\right)^{\xi_{t}}}\right)
      && h=1\text{ year},\ \beta_{\text{floor}}=0.1
  \end{align*}}%
  where the single index model gives $\alpha_{i}(t)$ and $\beta_{i}(t)$, $\tilde{g}_{M,t}$ is the 252-day
  exponential moving average of market growth, and the market state $\xi_{t}$ is positive when the market falls.

  \vspace{0.5em}
  The preference weight lies between $-1$ and $1$, with the sign of the expected growth rate.
\end{frame}

\begin{frame}{How the Market State Tilts the Preference Weight}
  \vspace{0.3em}
  The market state $\xi_{t}$ sets how the market sensitivity scales the expected growth rate:
  \begin{center}
    \includegraphics{slides/figs/Fig-L15a-Deck-Tilt.pdf}
  \end{center}
  At the same positive expected growth rate, the preference weight favors a low market sensitivity when the
  market falls and a high one when the market rises.
\end{frame}

\begin{frame}{The Cobb-Douglas Weights}
  \vspace{0.3em}
  Choose the share counts $n_{i}$ of the basket's tickers with $\gamma_{i}>0$, at the prices $S_{i}$
  (USD/share), to spend the wealth $W$ (USD):
  {\setlength{\jot}{2pt}
  \begin{align*}
    &\max_{n}\ \sum_{\gamma_{i}>0}\gamma_{i}\ln n_{i}\quad\text{subject to}\quad\sum_{i}n_{i}S_{i}=W
      && \text{utility and budget}\\
    &\gamma_{i}/n_{i}=\lambda S_{i}\quad\Longrightarrow\quad n_{i}S_{i}=\gamma_{i}/\lambda
      && \text{first-order condition, multiplier }\lambda\\
    &W=\sum_{\gamma_{j}>0}\gamma_{j}\big/\lambda && \text{add over the tickers}\\
    \Longrightarrow\quad&\boxed{w_{i}=\frac{n_{i}S_{i}}{W}=\frac{\max(\gamma_{i},0)}{\sum_{j}\max(\gamma_{j},0)}}
  \end{align*}}%
  For three tickers with $\gamma=(0.6,\ 0.3,\ -0.1)$, the weights are $(0.67,\ 0.33,\ 0.00)$. When no preference weight is positive, every
  weight is $0$, and the portfolio holds cash.
\end{frame}

\begin{frame}{Under the Trading Limits}
  \vspace{0.3em}
  At each decision, the engine chooses the share counts $n_{i}$ that maximize the Cobb-Douglas utility
  within the trading limits:
  \[
    U(n;\gamma)=\sum_{\gamma_{i}>0}\gamma_{i}\ln n_{i}
  \]
  \begin{itemize}\setlength{\itemsep}{0.5em}
    \item \hilite{Shared limits}: the weight, volatility, and trading-cost limits of every strategy apply. With the hedge on, so does the market sensitivity limit, and the engine can buy up to five tickers with zero or negative market sensitivity from its hedge pool.
    \item \hilite{Left out}: a ticker with $\gamma_{i}\leq 0$ holds no shares.
  \end{itemize}

  \vspace{0.3em}
  The cash stays at or above zero. When a limit stops the engine from reaching the Cobb-Douglas weights, the engine holds the best portfolio within the limits, and cash can remain.
\end{frame}
