% ==== lattice definition =====================================
\begin{frame}{Binomial Lattice Model}
  Let $S_0>0$ be today's share price and $\Delta t>0$ the step length
  in years. The move factors satisfy $u>d>0$, and the up probability is
  $0<p<1$. The share price after one step is given by:
  \[
    S_1=
    \begin{cases}
      S_0u, & \text{with probability }p,\\
      S_0d, & \text{with probability }1-p.
    \end{cases}
  \]

  \vspace{0.6em}
  The factors $u$, $d$, and probability $p$ stay fixed.
  Successive moves are independent.

  After $t$ steps, there are $t+1$ possible prices.

  \slidenote{Scope:}{we model share-price changes without dividend payments during the step.}
\end{frame}

% ==== terminal distribution ==================================
\begin{frame}{Price Distribution}
  After $t$ independent steps, let $k$ be the number of up moves.
  There are $t-k$ down moves, so the price is given by:
  \[
    \underbrace{S_{t,k}=S_0u^kd^{t-k}}_{\text{\footnotesize price after }t\text{\footnotesize steps}},
    \qquad k=0,\ldots,t.
  \]

  There are $\binom tk$ ways to arrange those moves.
  Under the real-world probability rule $\Pmeasure$, the node probability is given by:
  \[
    \Pmeasure(k\text{ up moves in }t\text{ steps})
    =\underbrace{\binom tk p^k(1-p)^{t-k}}_{\text{\footnotesize binomial probability}}.
  \]

  Different move orders can reach the same price.
\end{frame}



\begin{frame}{Example: Recombination}
  The textbook check uses $S_0=20$, $u=1.1$, $d=0.9$, and $p=0.6523$.
  An up move followed by a down move reaches the same price as the reverse.

  This common price is given by:
  \[
    \underbrace{20(1.1)(0.9)}_{\text{up then down}}
    =\underbrace{20(0.9)(1.1)}_{\text{down then up}}
    =19.8.
  \]

  Each path has probability $p(1-p)$, so the shared node has probability
  $2p(1-p)$.

  \vspace{0.7em}
  This is \hilite{recombination}. It holds because $ud=du$.
  Here $ud=0.99$, so recombination does not require $ud=1$.

  \slidenote{Optional extension:}{the recombination notebook compares path counts and node counts.}
\end{frame}

% ==== real-world estimation ==================================
\begin{frame}{Estimate Parameters from Data}
  For forecasting, estimate $(u,d,p)$ from growth rates $g_2,\ldots,g_T$
  observed at intervals of $\Delta t$ years. Start two empty lists:
  up factors $U$ and down factors $D$.

  \vspace{0.5em}
  \begin{enumerate}\setlength{\itemsep}{0.55em}
    \item Append $e^{g_t\Delta t}$ to $U$ if $g_t>0$, or to $D$ if $g_t<0$.
      Skip zero-growth observations when building the lists.
    \item Compute the average factors using:
      \[
        u=\frac{1}{|U|}\sum_{v\in U}v,
        \qquad
        d=\frac{1}{|D|}\sum_{v\in D}v.
      \]
    \item Estimate the up probability as the fraction of observed steps that rise:
      \[
        p=\frac{|U|}{T-1}.
      \]
  \end{enumerate}

  \slidenote{Counts:}{$|U|$ and $|D|$ are list lengths. Repeated factors count separately. Zero-growth days stay in the denominator of $p$; both lists must be nonempty.}
\end{frame}

\begin{frame}{Check the Model Against New Data}
  Historical estimates give us a model for forecasting.

  \vspace{0.8em}
  \begin{itemize}\setlength{\itemsep}{0.9em}
    \item The estimates depend on which observations we use.
    \item Two move factors summarize many different observed price changes.
    \item Test the fitted model on later data: compare its predicted price
      range with the prices that actually occur.
  \end{itemize}

  \slidenote{Example:}{the companion notebook compares model-predicted coverage with observed 2025 coverage.}
\end{frame}





% ==== probability measures ===================================
\begin{frame}{Forecasting and Pricing}
  Forecasting asks which prices may occur. Pricing asks what a future
  payoff is worth today. These tasks use different probability rules:

  \vspace{0.9em}
  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.47\textwidth}
      {\large\textbf{\ink{Real-world $\Pmeasure$}}}
      \vspace{0.5em}
      \begin{itemize}\setlength{\itemsep}{0.7em}
        \item Estimate the up probability $p$ from data.
        \item Use it to forecast prices and measure risk.
      \end{itemize}
    \end{column}
    \begin{column}{0.47\textwidth}
      {\large\textbf{\ink{Risk-neutral $\Qmeasure$}}}
      \vspace{0.5em}
      \begin{itemize}\setlength{\itemsep}{0.7em}
        \item Calculate $q$ from the move factors and risk-free growth.
        \item Use it to value future payoffs.
      \end{itemize}
    \end{column}
  \end{columns}

  \vspace{1em}
  The pricing weight $q$ is not a forecast of the frequency of up moves.
\end{frame}

\begin{frame}{No-Arbitrage Condition}
  An \hilite{arbitrage} costs nothing initially, cannot lose money,
  and makes money in at least one future state.

  Let $g_y$ be the risk-free growth rate in inverse years.
  Over $\Delta t$ years, its accumulation factor is given by:
  \[
    R_f=e^{g_y\Delta t}.
  \]
  To avoid arbitrage, the factors must satisfy the condition:
  \[
    \underbrace{d<R_f<u}_{\text{\footnotesize no-arbitrage condition}}.
  \]

  \begin{itemize}\setlength{\itemsep}{0.5em}
    \item If $R_f\leq d$, borrow at the risk-free rate and buy the share.
    \item If $R_f\geq u$, sell a borrowed share and invest the proceeds.
  \end{itemize}
\end{frame}

\begin{frame}{Arbitrage: Two Examples}
  In each example, the share costs $S_0=100$ and the trade starts with
  no net cash investment.

  \vspace{0.8em}
  \textbf{\ink{Borrow and buy: $R_f=d=1.05$, $u=1.20$}}
  \begin{itemize}\setlength{\itemsep}{0.4em}
    \item Borrow 100 and buy one share. The loan repayment is 105.
    \item Sell for 105 or 120. Profit after repayment: 0 or 15.
  \end{itemize}

  \vspace{0.8em}
  \textbf{\ink{Sell short and invest: $R_f=u=1.10$, $d=0.90$}}
  \begin{itemize}\setlength{\itemsep}{0.4em}
    \item Sell a borrowed share for 100 and invest it. The investment becomes 110.
    \item Buy back the share for 110 or 90. Profit: 0 or 20.
  \end{itemize}

  \slidenote{Model assumptions:}{borrowing, lending, and short selling are allowed; transaction costs are ignored.}
\end{frame}

\begin{frame}{Calculate the Risk-Neutral Probability}
  With arbitrage ruled out, choose the risk-neutral up probability $q$
  so the weighted average price factor equals $R_f$.
  With down probability $1-q$, this condition is written as:
  \[
    qu+(1-q)d=R_f.
  \]
  Solving this equation for $q$ gives the pricing weight:
  \[
    \underbrace{q=\frac{R_f-d}{u-d}}_{\text{\footnotesize risk-neutral up probability}}.
  \]

  Since $d<R_f<u$, we have $0<q<1$.

  \slidenote{Meaning:}{$q$ is a pricing weight. The historical up probability remains $p$.}
\end{frame}

\begin{frame}{Value a Future Payoff}
  Suppose a claim pays $H_u$ dollars after an up move and $H_d$ dollars
  after a down move. Its no-arbitrage value today, $V_0$, is given by:
  \[
    V_0=
    \underbrace{\frac{1}{R_f}}_{\text{\footnotesize discount factor}}
    \underbrace{\left[qH_u+(1-q)H_d\right]}_{\text{\footnotesize weighted payoff}}.
  \]

  \vspace{0.8em}
  \begin{enumerate}\setlength{\itemsep}{0.7em}
    \item Weight the two payoffs using the risk-neutral probability $q$.
    \item Divide by $R_f$ to discount back one step.
  \end{enumerate}

  \slidenote{Later in the course:}{we derive this pricing rule when we study options.}
\end{frame}

% ==== stylized facts =========================================
\begin{frame}{Test the Lattice Against Market Data}
  We now have a pricing rule. But how well does the underlying model
  describe market data? Return to the three patterns from L3a:

  \vspace{0.8em}
  \begin{itemize}\setlength{\itemsep}{0.8em}
    \item \hilite{Heavy tails}: large growth rates occur more often than a
      fitted Normal model predicts.
    \item \hilite{Weak linear autocorrelation}: growth rates on different
      days have little linear relationship beyond short lags.
    \item \hilite{Volatility clustering}: large movements tend to occur
      together, as do small movements.
  \end{itemize}

  \slidenote{Probability rule:}{use real-world $\Pmeasure$ to compare the model with market data. See \href{https://doi.org/10.1080/713665670}{Cont (2001)} for the empirical patterns.}
\end{frame}

\begin{frame}{Growth-Rate Distribution}
  To make that comparison, express the lattice moves as growth rates.
  At step $j$, the price factor is $u$ or $d$. For a step of length
  $\Delta t$ years, the growth rate is given by:
  \[
    g_j=\frac{1}{\Delta t}\ln\!\left(\frac{S_j}{S_{j-1}}\right)
    =
    \begin{cases}
      \ln u/\Delta t, & \text{with probability }p,\\
      \ln d/\Delta t, & \text{with probability }1-p.
    \end{cases}
  \]

  \vspace{0.7em}
  The model produces only two possible growth rates, both in inverse years.

  Changing $p$ changes how often each value occurs, not the values themselves.
\end{frame}

\begin{frame}{One-Step Mean and Variance}
  Keep $u>d>0$, $0<p<1$, and $\Delta t>0$ years fixed.

  The mean growth rate $\mu_g$ under $\Pmeasure$ is given by:
  \[
    \mu_g\equiv\mathbb E_{\Pmeasure}[g_j]
    =\underbrace{\frac{p\ln u+(1-p)\ln d}{\Delta t}}_{\text{\footnotesize mean growth rate}}.
  \]

  The variance $\sigma_g^2$ is given by:
  \[
    \sigma_g^2\equiv\operatorname{Var}_{\Pmeasure}(g_j)
    =\underbrace{\frac{p(1-p)}{(\Delta t)^2}
      \left[\ln\!\left(\frac ud\right)\right]^2}_{\text{\footnotesize growth-rate variance}}.
  \]

  Both quantities stay fixed because $u$, $d$, and $p$ stay fixed.
\end{frame}



\begin{frame}{Heavy Tails}
  The two growth rates have a maximum magnitude.
  This bound, $M$, is given by:
  \[
    M=\frac{\max\{|\ln d|,|\ln u|\}}{\Delta t}.
  \]
  Since every one-step growth rate satisfies $|g_j|\leq M$,
  the probability of exceeding this bound is given by:
  \[
    \underbrace{\Pmeasure(|g_j|>x)=0\quad\text{for }x>M}_{\text{\footnotesize no observations beyond the bound}}.
  \]

  \vspace{0.6em}
  The fixed lattice cannot produce heavy-tailed one-step growth rates.
\end{frame}

\begin{frame}{Temporal Dependence}
  Each growth rate depends on its own up or down move.
  Since successive moves are independent, so are growth rates at
  different steps.

  Let $\tau\geq1$ be an integer lag. Their covariance is given by:
  \[
    \operatorname{Cov}_{\Pmeasure}(g_j,g_{j+\tau})=0.
  \]
  Since the growth-rate variance $\sigma_g^2$ is positive, the ACF is given by:
  \[
    \underbrace{\rho_g(\tau)=
      \frac{\operatorname{Cov}_{\Pmeasure}(g_j,g_{j+\tau})}{\sigma_g^2}
      =0}_{\text{\footnotesize zero autocorrelation}}.
  \]

  \slidenote{Interpretation:}{the model matches weak growth-rate autocorrelation because we assumed independent moves.}
\end{frame}



\begin{frame}{Volatility Clustering}
  Independent moves also give independent growth-rate magnitudes.
  For any integer lag $\tau\geq1$, their covariance is given by:
  \[
    \operatorname{Cov}_{\Pmeasure}(|g_j|,|g_{j+\tau}|)=0.
  \]

  Let $\mathcal F_j$ represent the information in the first $j$ moves.
  Given that history, the future growth-rate variance is still given by:
  \[
    \underbrace{\operatorname{Var}_{\Pmeasure}
      (g_{j+\tau}\mid\mathcal F_j)=\sigma_g^2}_{\text{\footnotesize constant conditional variance}}.
  \]

  A large move today does not make a large move tomorrow more likely.
  The fixed lattice has no volatility clustering.
\end{frame}

\begin{frame}{The Lattice Passes One of Three Tests}
  Compare the fixed lattice with the market patterns from L3a:

  \vspace{0.9em}
  \begin{itemize}\setlength{\itemsep}{0.9em}
    \item \textbf{\ink{Heavy tails: fails.}}
      Only two one-step growth rates are possible.
    \item \textbf{\ink{Weak growth-rate autocorrelation: passes.}}
      Independent moves give zero autocorrelation.
    \item \textbf{\ink{Volatility clustering: fails.}}
      Past movements do not change future variance.
  \end{itemize}

  \vspace{0.8em}
  The lattice is a useful baseline, but these assumptions limit its
  ability to describe market data.
\end{frame}
