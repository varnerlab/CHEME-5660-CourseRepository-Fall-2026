\begin{frame}{The Binomial Lattice Model}
  Let $S_0>0$ be today's share price. At each step, multiply the price by a
  constant up factor $u>0$ with probability $p$, or a constant down factor
  $d>0$ with probability $1-p$, where $u>d$.

  \vspace{0.5em}
  After $t$ steps, let $K_t$ count the up moves. Then the node indexed by
  $K_t=k$ has price and real-world probability:
  \[
    \underbrace{S_{t,k}=S_0u^kd^{t-k}}_{\text{node price}},
    \qquad
    \underbrace{\Pmeasure(K_t=k)=\binom tk p^k(1-p)^{t-k}}_{\text{node probability}}.
  \]

  The model is transparent, but fixed $u$, $d$, and $p$ do not reproduce heavy
  tails or volatility clustering.
\end{frame}

\begin{frame}{A Forecast Is Not Yet a Trading Rule}
  A price model describes possible states and their probabilities. A trading
  rule must add the decisions that define the event of interest.

  \vspace{0.7em}
  \begin{itemize}\setlength{\itemsep}{0.6em}
    \item When do we enter, and how many shares do we buy?
    \item Is the exit scheduled at a terminal date or monitored before it?
    \item Which entry and exit prices represent executable prices?
    \item What benchmark return and costs must the trade clear?
    \item Is success defined by a strict or non-strict target?
  \end{itemize}

  \vspace{0.5em}
  Without these choices, ``probability of profit'' is not well defined.
\end{frame}

\begin{frame}{Net Present Value of a Long Position}
  Buy $n_0>0$ shares at price $S_0$ at time 0, then sell all shares at price
  $S_T$ at time $T=N\Delta t$. For annual continuously compounded benchmark
  growth rate $g_b$, the frictionless time-0 NPV is:
  \[
    \underbrace{\operatorname{NPV}(g_b,T)}_{\text{time-0 dollars}}
    =\underbrace{-n_0S_0}_{\text{entry now}}
    +\underbrace{n_0S_Te^{-g_bT}}_{\text{discounted exit}}.
  \]

  Divide by the initial share cost to obtain a dimensionless present-value return:
  \[
    \underbrace{\rho_T}_{\text{scaled NPV}}
    :=\frac{\operatorname{NPV}(g_b,T)}{n_0S_0}
    =e^{-g_bT}\frac{S_T}{S_0}-1.
  \]

  A positive price change alone need not clear the benchmark, spread, and costs.
\end{frame}

\begin{frame}{Short Holding Period}
  When $|g_b|T$ is small, $e^{-g_bT}\approx1$. The scaled NPV is then
  approximately the fractional change in share price:
  \[
    \underbrace{\rho_T}_{\text{scaled NPV}}
    \approx \underbrace{\frac{S_T}{S_0}-1}_{\text{price return}}.
  \]

  Under the binomial lattice, $K_N\sim\operatorname{Binomial}(N,p)$ and:
  \[
    \frac{S_T}{S_0}=u^{K_N}d^{N-K_N}.
  \]

  Therefore the $N+1$ terminal nodes give the complete discrete distribution of
  the approximate return.
\end{frame}

\begin{frame}{Longer Holding Period}
  For a longer horizon, or whenever $|g_b|T$ is not small, retain the benchmark
  discount factor. At terminal node $K_N=k$:
  \[
    \underbrace{\rho_T(k)}_{\text{present-value return}}
    =\underbrace{e^{-g_b N\Delta t}}_{\text{discount}}
      \underbrace{u^kd^{N-k}}_{\text{price ratio}}-1,
    \qquad k=0,\ldots,N.
  \]

  The node probability remains:
  \[
    \Pmeasure(K_N=k)=\binom Nk p^k(1-p)^{N-k}.
  \]

  The benchmark changes the return attached to each node. It does not change the
  real-world node probabilities specified by $p$. The matching one-step gross
  benchmark return from L3b is $R_b=e^{g_b\Delta t}$.
\end{frame}

\begin{frame}{Cumulative Probability of a Terminal Target}
  Let $\rho_\star>-1$ be a dimensionless target. We ask for the probability that
  the scheduled terminal return \hilite{strictly exceeds} this target:
  \[
    \Pmeasure\bigl(\rho_T(K_N)>\rho_\star\bigr).
  \]

  Since $u>d>0$, the return $\rho_T(k)$ increases with the up-move count $k$.
  Therefore, the target event is equivalent to:
  \[
    K_N>\tau(\rho_\star)
  \]
  for a real-valued threshold $\tau(\rho_\star)$. The probability is a binomial tail,
  not the probability of one terminal node.
\end{frame}

\begin{frame}{Solve the Return Inequality for the Up Count}
  Begin with the strict target at a node containing $k$ up moves:
  \[
    e^{-g_b N\Delta t}u^kd^{N-k}-1>\rho_\star.
  \]

  Move positive terms, take logarithms, and use $\log(u/d)>0$:
  \[
    k\log(u/d)>\log(1+\rho_\star)+g_b N\Delta t-N\log d.
  \]

  Thus the real-valued threshold is:
  \[
    \underbrace{\tau(\rho_\star)}_{\text{count threshold}}
    =\frac{\log(1+\rho_\star)+g_b N\Delta t-N\log d}{\log(u/d)}.
  \]
\end{frame}

\begin{frame}{Strict Targets Need the Next Integer}
  The event is $K_N>\tau(\rho_\star)$. Because $K_N$ is integer valued, the smallest
  successful up count is:
  \[
    \underbrace{k_{\min}}_{\text{first successful count}}
    =\lfloor\tau(\rho_\star)\rfloor+1.
  \]

  With $K_N\sim\operatorname{Binomial}(N,p)$, the target probability is:
  \[
    \boxed{\Pmeasure(\rho_T>\rho_\star)
    =\sum_{k=\max(0,k_{\min})}^N\binom Nk p^k(1-p)^{N-k}.}
  \]

  A non-strict event $\rho_T\ge \rho_\star$ uses a different integer boundary when a
  node lies exactly on the target.
\end{frame}

\begin{frame}{Threshold Edge Cases}
  Keep the true integer $k_{\min}$ instead of clamping it into the feasible range.

  \vspace{0.7em}
  \begin{itemize}\setlength{\itemsep}{0.65em}
    \item If $k_{\min}\le0$, every feasible terminal node succeeds, so the
      probability is 1.
    \item If $k_{\min}>N$, no feasible terminal node succeeds, so the probability is 0.
    \item At fixed $N$, increasing $g_b$ raises the threshold and cannot
      increase the tail probability.
    \item Changing the horizon changes both the threshold and the distribution
      of $K_N$. Recompute the full probability.
  \end{itemize}

  \slidenote{Implementation:}{for a strict tail, use the complementary binomial CDF at $k_{\min}-1$ only after handling the two edge cases.}
\end{frame}

\begin{frame}{Terminal Rules and First-Passage Rules Are Different}
  The binomial tail concerns liquidation at the scheduled date $T$. A take-profit
  or stop-loss rule monitored at each step depends on the \hilite{first} boundary
  crossing and is path dependent.

  \vspace{0.6em}
  \begin{itemize}\setlength{\itemsep}{0.55em}
    \item Two paths can end at the same terminal node but cross a boundary at
      different earlier times.
    \item Propagate only probability mass that has not already crossed a boundary,
      then record the absorbed mass at each first crossing.
    \item A stop order becomes a market order after its trigger and need not fill
      at the trigger price. A limit order may not fill.
  \end{itemize}

  Terminal node probabilities alone cannot evaluate a monitored exit rule.
\end{frame}

\begin{frame}{N-Ary Lattice Models}
  Replace the two branch factors with $m$ positive factors
  $f_1,\ldots,f_m$. Let branch $j$ occur with probability $p_j\ge0$, where
  $\sum_{j=1}^mp_j=1$.

  \vspace{0.6em}
  After $t$ steps, a recombining node is indexed by a branch-count vector:
  \[
    \mathbf{x}=(x_1,\ldots,x_m),
    \qquad x_j\in\mathbb{Z}_{\ge0},
    \qquad \sum_{j=1}^mx_j=t.
  \]

  Order does not matter because the branch factors are constant and multiplication
  commutes. Paths with the same count vector share one node.
\end{frame}

\begin{frame}{N-Ary Node Values}
  For count vector $\mathbf{x}$ at level $t$, the node price is:
  \[
    \underbrace{S_t(\mathbf{x})}_{\text{node price}}
    =S_0\prod_{j=1}^mf_j^{x_j}.
  \]

  Independent branch draws give the multinomial node probability:
  \[
    \underbrace{\Pmeasure(\mathbf{X}=\mathbf{x})}_{\text{node probability}}
    =\frac{t!}{x_1!\cdots x_m!}\prod_{j=1}^mp_j^{x_j}.
  \]

  Distinct count vectors can still produce the same price if the factors have
  additional multiplicative relationships.
\end{frame}

\begin{frame}{More Branches Create More States}
  The number of nonnegative count vectors at level $t$ is the stars-and-bars count:
  \[
    \underbrace{L_t}_{\text{nodes at level }t}=\binom{t+m-1}{m-1}.
  \]

  The total number of nodes through height $h$ is:
  \[
    \underbrace{\textstyle\sum_{r=0}^{h}L_r}_{\text{all stored nodes}}
    =\binom{h+m}{m}.
  \]

  More branches give a finer discrete one-step distribution, but also increase
  the state count. They do not automatically improve out-of-sample forecasts.
\end{frame}

\begin{frame}{N-Ary Example: Discretize, Build, Check}
  The companion notebook uses empirical log growth rates for one firm.

  \vspace{0.6em}
  \begin{enumerate}\setlength{\itemsep}{0.6em}
    \item Divide the one-step growth sample into three equal-width bins.
    \item Estimate one representative factor and one frequency for each bin.
    \item Build each recombining state from its three branch counts.
    \item Compute its multinomial probability and verify that all node
      probabilities at the selected level sum to one.
    \item Plot price and annualized multistep log growth against probability mass.
  \end{enumerate}

  \slidenote{Interpretation:}{the histogram of node locations is not a probability distribution unless each node is weighted by its probability mass.}
\end{frame}
