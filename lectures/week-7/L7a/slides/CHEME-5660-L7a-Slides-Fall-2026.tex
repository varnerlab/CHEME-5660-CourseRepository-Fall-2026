\begin{frame}{Concept Review: The SIM Covariance}
  In the SIM $g_i=\alpha_i+\beta_i\,g_M+\varepsilon_i$, residuals have zero mean and are
  uncorrelated with the market and across assets, so $\E[g_i]=\alpha_i+\beta_i\,\E[g_M]$.
  \hilite{On the diagonal}:
  \[
    \begin{aligned}
      \Var(g_i)&=\beta_i^2\Var(g_M)+2\beta_i\underbrace{\Cov(g_M,\varepsilon_i)}_{=0}+\Var(\varepsilon_i)
        &&\quad\text{expand the SIM}\\
      &=\beta_i^2\,\sigma^2_{g,M}+\sigma^2_{g,\varepsilon,i}
        &&\quad\text{second assumption}
    \end{aligned}
  \]
  \hilite{Off the diagonal} ($i\neq j$), the second and third assumptions remove three terms:
  \[
    \begin{aligned}
      \Cov(g_i,g_j)&=\beta_i\beta_j\Var(g_M)+\beta_i\underbrace{\Cov(g_M,\varepsilon_j)}_{=0}
        +\beta_j\underbrace{\Cov(\varepsilon_i,g_M)}_{=0}+\underbrace{\Cov(\varepsilon_i,\varepsilon_j)}_{=0}\\
      &=\beta_i\beta_j\,\sigma^2_{g,M}
    \end{aligned}
  \]
\end{frame}

\begin{frame}{Concept Review: SIM Portfolio Inputs}
  With the betas in $\bbe$, split each entry into its market and residual parts:
  \[
    \symbf{\Sigma}_g=\sigma^2_{g,M}\begin{bmatrix}\beta_1^2&\beta_1\beta_2&\cdots\\ \beta_2\beta_1&\beta_2^2&\cdots\\ \vdots&\vdots&\ddots\end{bmatrix}
    +\begin{bmatrix}\sigma^2_{g,\varepsilon,1}&0&\cdots\\ 0&\sigma^2_{g,\varepsilon,2}&\cdots\\ \vdots&\vdots&\ddots\end{bmatrix}
    =\underbrace{\sigma^2_{g,M}\,\bbe\bbe^{\top}}_{\text{market}}+\underbrace{\Dg}_{\text{residual}}
  \]
  Substituting the training-data estimates gives the SIM inputs:
  \[
    \boxed{\gSIM=\hat{\bal}+\hat{\bbe}\,g'_M,\qquad
      \SigSIM=s^2_{g,M}\,\hat{\bbe}\hat{\bbe}^{\top}+\hat{\mathbf D}_g}
  \]
  \begin{itemize}\setlength{\itemsep}{0.2em}
    \item \hilite{Same means}: a least-squares line with an intercept passes
      through the sample means, so $\hat\alpha_i+\hat\beta_i\,g'_M=g'_i$.
    \item \hilite{Different covariance}: off the diagonal, the SIM covariance is
      the sample covariance with the residual covariances set to zero.
  \end{itemize}
\end{frame}

\begin{frame}{Portfolio Risk under a SIM}
  For fixed weights $\bw$, substituting the SIM entries of $\symbf{\Sigma}_g$ into
  the portfolio variance gives:
  \[
    \begin{aligned}
      \Var(g_p)&=\textstyle\sum_{i}\sum_{j}w_iw_j\Cov(g_i,g_j)
        &&\text{portfolio variance (L6a)}\\
      &=\textstyle\sum_{i}w_i^2\bigl(\beta_i^2\sigma^2_{g,M}+\sigma^2_{g,\varepsilon,i}\bigr)
        +\sum_{i}\sum_{j\neq i}w_iw_j\beta_i\beta_j\sigma^2_{g,M}
        &&\text{SIM entries}\\
      &=\textstyle\sigma^2_{g,M}\sum_{i}\sum_{j}(w_i\beta_i)(w_j\beta_j)
        +\sum_{i}w_i^2\sigma^2_{g,\varepsilon,i}
        &&\text{collect market terms}\\
      &=\textstyle\sigma^2_{g,M}\bigl(\sum_{i}w_i\beta_i\bigr)^2
        +\sum_{i}w_i^2\sigma^2_{g,\varepsilon,i}
        &&\text{a perfect square}\\
      &=\underbrace{\beta_p^2\,\sigma^2_{g,M}}_{\text{market risk}}
        +\underbrace{\textstyle\sum_{i}w_i^2\sigma^2_{g,\varepsilon,i}}_{\text{residual risk}}
        &&\beta_p=\textstyle\sum_{i}w_i\beta_i
    \end{aligned}
  \]
  Market risk depends on $\bw$ only through $\beta_p$. Even weights over many
  assets with uncorrelated, similar-size residuals drive residual risk toward zero.
\end{frame}

\begin{frame}{The Long-Only Problem with SIM Inputs}
  SIM-1 is L6a's long-only problem with SIM inputs and a floor $g_\star$ (year$^{-1}$):
  \[
    \boxed{
    \begin{aligned}
      \underset{\bw}{\text{minimize}}\quad
        &\underbrace{s^2_{g,M}\bigl(\hat{\bbe}^{\top}\bw\bigr)^2}_{\text{market risk}}
        +\underbrace{\textstyle\sum_{i}s^2_{g,\varepsilon,i}\,w_i^2}_{\text{residual risk}}
        &&\quad\text{minimize }\bw^{\top}\SigSIM\bw\\
      \text{subject to}\quad &\gSIM^{\top}\bw\ge g_\star
        &&\quad\text{meet or exceed the growth target}\\
      &\bone^{\top}\bw=1
        &&\quad\text{weights sum to one}\\
      &0\le w_i\le1,\quad i\in\mathcal P
        &&\quad\text{exclude short positions}.
    \end{aligned}}
  \]
  Sweeping $g_\star$ up from the long-only GMV growth rate traces the efficient
  frontier. We compare both allocations under one covariance, then on 2025 prices.

\end{frame}

\begin{frame}{Adding a Risk-Free Asset}
  A \hilite{risk-free asset} has a known growth rate $g_f$ (year$^{-1}$) and zero
  variance over the holding period, such as a Treasury bill held to maturity:

  \begin{columns}[c]
    \begin{column}{0.56\textwidth}
      \includegraphics[width=\textwidth]{../figs/capital-allocation-line/cal.pdf}
    \end{column}
    \begin{column}{0.42\textwidth}
      \begin{itemize}\setlength{\itemsep}{0.5em}
        \item \hilite{Capital allocation line (CAL)}: mixing the risk-free asset
          with a risky portfolio gives a straight line from $(0,g_f)$ through
          that portfolio.
        \item \hilite{Tangent portfolio} $\Tan$: the steepest such line touches
          the efficient frontier at $\Tan$.
        \item \hilite{Lending and borrowing}: points before $\Tan$ lend at $g_f$,
          and points beyond it borrow at $g_f$.
      \end{itemize}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{The Capital Allocation Line}
  A \hilite{complete portfolio} $c$ holds $w_f$ in the risk-free asset and $1-w_f$
  in a risky portfolio $p$:
  \[
    \begin{aligned}
      \E[g_c]&=w_f\,g_f+(1-w_f)\,\E[g_p] &&\text{expectation is linear}\\
      &=g_f+(1-w_f)\bigl(\E[g_p]-g_f\bigr) &&\text{regroup}\\
      \sigma^2_{g,c}&=w_f^2\underbrace{\Var(g_f)}_{=0}+2w_f(1-w_f)\underbrace{\Cov(g_f,g_p)}_{=0}+(1-w_f)^2\sigma^2_{g,p}
        &&g_f\text{ is constant}\\
      \sigma_{g,c}&=|1-w_f|\,\sigma_{g,p} &&\text{take the root}
    \end{aligned}
  \]
  For $w_f\le1$ and $\sigma_{g,p}>0$, substituting $1-w_f=\sigma_{g,c}/\sigma_{g,p}$
  gives the \hilite{CAL}:
  \[
    \boxed{\E[g_c]=g_f+\SR_p\,\sigma_{g,c},\qquad
      \SR_p=\bigl(\E[g_p]-g_f\bigr)/\sigma_{g,p}\ \text{(Sharpe ratio)}}
  \]
  We lend for $0<w_f<1$, hold $p$ alone for $w_f=0$, and borrow for $w_f<0$.
  \slidenote{Daily data:}{the slope is a one-day Sharpe ratio. With independent increments,
    $\SR_p^{\mathrm{annual}}=\sqrt{252}\,\SR_p$
    (\href{https://web.stanford.edu/~wfsharpe/art/sr/SR.htm}{Sharpe, 1994}).}
\end{frame}

\begin{frame}{The Tangent Portfolio}
  The \hilite{tangent portfolio} $\Tan$ maximizes the Sharpe ratio. Let $\bmu$ hold the
  expected growth rates, $\Sig$ be positive definite, and the GMV portfolio
  (shorts allowed) grow faster than $g_f$. With $\be=\bmu-g_f\bone$, solve:
  \[
    \underset{\bw}{\text{maximize}}\quad\frac{\be^{\top}\bw}{\sqrt{\bw^{\top}\Sig\bw}}
    \qquad\text{subject to}\quad\bone^{\top}\bw=1
    \tag{T-1}
  \]
  Scaling $\bw$ leaves the ratio unchanged, so we set its gradient to zero
  (T-2, \href{\repo/advanced/tangent-derivation/CHEME-5660-L7a-Derivation-Tangent-Fall-2026.ipynb}{derivation}) and rescale. The weights point along $\Sig^{-1}\be$, and summing to one gives:
  \[
    \boxed{\bw_{\Tan}=\frac{\Sig^{-1}\left(\bmu-g_f\bone\right)}
      {\bone^{\top}\Sig^{-1}\left(\bmu-g_f\bone\right)}}
    \tag{T-3}
  \]
  When T-3 has negative weights, the long-only tangent portfolio has no closed form.
\end{frame}

\begin{frame}{Two-Fund Separation}
  The steepest CAL separates choosing a risky portfolio from choosing how much
  risk to take. Mean-variance investors with the same inputs who can lend and
  borrow freely at $g_f$ hold the \hilite{same risky fund} $\Tan$, combined with the risk-free asset.
  Their risk preferences set $w_f$
  (\href{https://doi.org/10.2307/2296205}{Tobin, 1958}).

  \vspace{0.6em}
  Real investors face limits on how they finance their positions:
  \begin{itemize}\setlength{\itemsep}{0.5em}
    \item \hilite{Long-only weights}: the risky fund is the long-only
      maximum-Sharpe portfolio, and investors still choose $w_f$.
    \item \hilite{No borrowing} ($w_f\ge0$): the efficient boundary follows the
      CAL up to $\Tan$, then the risky efficient frontier.
    \item \hilite{Costlier borrowing}: a borrowing rate above $g_f$ gives
      borrowers a flatter line through a different risky portfolio.
  \end{itemize}
\end{frame}

\begin{frame}{Risky and Risk-Free Allocation with SIM Inputs}
  Choose the risky weights $\bw$ and put the rest, $w_f=1-\bone^{\top}\bw$, in
  the risk-free asset. For a target $g_\star>g_f$ (year$^{-1}$), problem SIM-2 is:
  \[
    \boxed{
    \begin{aligned}
      \underset{\bw}{\text{minimize}}\quad &\bw^{\top}\SigSIM\bw
        &&\quad\text{minimize growth-rate variance}\\
      \text{subject to}\quad
        &\gSIM^{\top}\bw+\underbrace{\bigl(1-\bone^{\top}\bw\bigr)}_{w_f}g_f\ge g_\star
        &&\quad\text{meet or exceed the growth target}\\
      &0\le w_i\le1,\quad i\in\mathcal P
        &&\quad\text{exclude short positions}.
    \end{aligned}}
  \]
  The fraction $w_f$ has no sign constraint, so $w_f<0$ means borrowing at $g_f$.

  \vspace{0.4em}
  Eliminating $w_f$ gives the excess-growth constraint
  $(\gSIM-g_f\bone)^{\top}\bw\ge g_\star-g_f$.
\end{frame}

\begin{frame}{The Solutions Lie on a Ray}
  Scaling $\bw$ by $c>0$ scales the excess growth by $c$ and the variance by
  $c^2$. So, until a bound $w_i\le1$ binds, every target selects the same risky
  direction:
  \[
    \boxed{\bw(g_\star)=(1-w_f)\,\bw_{\Tan},\qquad
      1-w_f=\frac{g_\star-g_f}{\gSIM^{\top}\bw_{\Tan}-g_f}}
    \tag{SIM-3}
  \]
  where the long-only tangent portfolio, $\bw_{\Tan}=\bw/\bone^{\top}\bw$, is
  the solution at any target, rescaled to sum to one.

  \vspace{0.5em}
  \begin{itemize}\setlength{\itemsep}{0.5em}
    \item Targets below $\gSIM^{\top}\bw_{\Tan}$ lend ($0<w_f<1$).
    \item Targets above it borrow ($w_f<0$).
  \end{itemize}

\end{frame}

\begin{frame}{Summary}
\begin{itemize}\setlength{\itemsep}{0.9em}
\item \hilite{The covariance model changes the allocation}: the SIM keeps the
sample means but sets the residual covariances to zero. Under the sample
covariance, the SIM weights carry at least as much risk at every growth floor.
\item \hilite{The tangent portfolio sets the steepest CAL}: the Sharpe ratio is
the line's slope, and maximizing it selects the tangent portfolio.
\item \hilite{Lending and borrowing set the complete portfolio's risk}: investors
who can lend and borrow freely at the risk-free rate hold the same risky fund.
Financing limits can change that fund.
\end{itemize}
\end{frame}
