% ==== concept review =========================================
\begin{frame}{Concept Review: Utility Functions}
  Utility functions describe a decision-maker's preferences over choices, outcomes,
  or states of the world. A larger utility is always better.

  \vspace{0.5em}
  An agent chooses from alternatives $\mathcal{X}=\{x_1,\dots,x_n\}$. A
  \hilite{utility function} $U:\mathcal{X}\rightarrow\mathbb{R}$ assigns each one a
  number, in \emph{utils}, that ranks the agent's preferences:
  \[
    \boxed{U(x_i)>U(x_j)\;\Longleftrightarrow\;x_i\succ x_j
      \qquad\text{and}\qquad
      U(x_i)=U(x_j)\;\Longleftrightarrow\;x_i\sim x_j}
  \]
  \begin{itemize}\setlength{\itemsep}{0.4em}
    \item \hilite{Ordinal}: $U$ ranks alternatives without measuring how much
      better one is, so any order-preserving transformation (e.g., $\ln U$ for
      $U>0$) keeps the same ranking.
    \item \hilite{Key idea}: a rational decision-maker always chooses the
      alternative with the largest utility.
  \end{itemize}
\end{frame}

\begin{frame}{Common Utility Functions}
  Utility functions, and their parameters, are subjective. Five common forms for
  a bundle of goods $\bx=(x_1,\dots,x_n)$ with $x_i\ge0$ and $\alpha_i>0$:

  \vspace{0.4em}
  {\small\setlength{\tabcolsep}{3pt}
  \begin{tabular}{@{}lll@{}}
    \toprule
    \textbf{Utility function} & \textbf{Form} & \textbf{Models}\\
    \midrule
    Linear & $U(\bx)=\sum_i\alpha_ix_i$ & goods that are perfect substitutes\\[0.2em]
    Logarithmic ($\beta>0$) & $U(\bx)=\ln\bigl(\sum_i\alpha_ix_i+\beta\bigr)$ & diminishing satisfaction from more goods\\[0.2em]
    Cobb--Douglas & $U(\bx)=\prod_ix_i^{\alpha_i}$ & goods held together\\[0.2em]
    Leontief & $U(\bx)=\min\left(x_1/\alpha_1,\dots,x_n/\alpha_n\right)$ & perfect complements, fixed proportions\\[0.2em]
    CES ($\eta>0$, $\eta\ne1$) & $U(\bx)=\Bigl(\sum_i\alpha_ix_i^{(\eta-1)/\eta}\Bigr)^{\eta/(\eta-1)}$ & substitution set by the elasticity $\eta$\\
    \bottomrule
  \end{tabular}}

  \vspace{0.8em}
  We use the Cobb--Douglas and CES utilities next. As $\eta$ changes,
  CES moves between the linear ($\eta\to\infty$), Cobb--Douglas ($\eta\to1$, with
  $\sum_i\alpha_i=1$), and Leontief ($\eta\to0$) forms.
\end{frame}

% ==== utility over holdings ==================================
\begin{frame}{Cobb–Douglas Utility over Holdings}
  An investor chooses share counts to maximize a \hilite{holding utility}, which has
  no variance term. Risk enters through the \hilite{preference weights}
  $\gamma_i\in(-1,1)$. With them as exponents, the Cobb--Douglas problem is:
  \[
    \boxed{\begin{aligned}
      \operatorname*{maximize}_{n_1,\ldots,n_{\np}}\quad&U(n_1,\ldots,n_{\np})=\prod_{i\in\mathcal{P}}n_i^{\gamma_i}\\
      \text{subject to}\quad&\sum_{i\in\mathcal{P}}n_i\,S_i(t)+\text{cash}(t)=\WP(t)&&\quad\text{(budget constraint)}\\
      &n_i\ge n_{\min}\quad\forall\,i\in\mathcal{P},\qquad\text{cash}(t)\ge0&&\quad\text{(share floor, no borrowing)}
    \end{aligned}}
  \]
  where $\mathcal{P}$ is the asset set, $n_i$ the share count of asset $i$, $S_i(t)$ its price (USD per share),
  $\WP(t)$ the wealth, and $\text{cash}(t)$ the uninvested wealth (USD). The floor is
  $n_{\min}>0$ shares, and cash adds no utility.
\end{frame}

\begin{frame}{The Cobb–Douglas Allocation}
  Assets with $\gamma_i>0$ form the \hilite{preferred set} $\Ap$, the
  \hilite{basket}. The rest form $\Am=\{i:\gamma_i\le0\}$. The floor keeps $U$ finite
  when $\gamma_i<0$.

  \vspace{0.4em}
  With positive prices, enough wealth to fund the floors, and a nonempty basket,
  non-preferred assets stay at their floors and cash is zero, which leaves the
  \hilite{net budget}:
  \[
    \Wadj=\WP(t)-n_{\min}\sum_{k\in\Am}S_k(t)
  \]
  When preferred holdings stay above their floors, the allocation is:
  \[
    \boxed{n_i^\star=\underbrace{\frac{\gamma_i}{\sum_{j\in\Ap}\gamma_j}}_{\text{preference share}}
      \cdot\underbrace{\frac{\Wadj}{S_i(t)}}_{\text{budget in shares}}\quad(i\in\Ap),
      \qquad n_k^\star=n_{\min}\quad(k\in\Am)}
  \]
  \slidenote{Derivation:}{\href{\repo/advanced/cobb-douglas-derivation/CHEME-5660-L8b-Derivation-CobbDouglas-Fall-2026.ipynb}{Cobb--Douglas companion},
    including what to do when a preferred holding reaches its floor.}
\end{frame}

\begin{frame}{CES Utility: A Concentration Dial}
  CES adds the elasticity $\eta>0$, which sets how willingly the allocator trades
  one preferred asset for another. For $\eta\ne1$:
  \[
    U_{\mathrm{CES}}=\Bigl(\sum_{i\in\Ap}\gamma_i\,n_i^{(\eta-1)/\eta}\Bigr)^{\eta/(\eta-1)}
    \qquad
    \boxed{n_i^\star=\Wadj\;\frac{\bigl(\gamma_i/S_i(t)\bigr)^{\eta}}
      {\sum_{j\in\Ap}S_j(t)\bigl(\gamma_j/S_j(t)\bigr)^{\eta}}\quad(i\in\Ap)}
  \]
  Share counts above the floors are proportional to the \emph{bang for the buck}
  $\gamma_i/S_i(t)$ raised to $\eta$. Three limits:
  \begin{itemize}\setlength{\itemsep}{0.35em}
    \item \hilite{Unit elasticity} ($\eta\to1$): the Cobb--Douglas allocation.
    \item \hilite{Large elasticity} ($\eta\to\infty$): the unique largest
      $\gamma_i/S_i(t)$ gets the whole budget above the floors. Every other asset falls
      to its floor.
    \item \hilite{Small elasticity} ($\eta\to0^+$): equal share counts of every
      preferred asset.
  \end{itemize}
  \slidenote{Proofs:}{\href{\repo/advanced/adaptive_utility/CHEME-5660-L8b-Advanced-CES-Limits-Fall-2026.ipynb}{CES companion},
    including why CES dollar weights depend on share prices when $\eta\ne1$.}
\end{frame}

% ==== preference weights =====================================
\begin{frame}{Preference Weights from the Single Index Model}
  Each trading day, we compute $\gamma_i(t)$ from the SIM parameters $(\alpha_i,\beta_i)$
  and two SPY inputs: the recent market growth rate $\tilde g_{M,t}$ and the
  market-state signal $\xi_t$, which says whether the market has been rising or falling:
  \[
    \boxed{\gamma_i(t)=\tanh\!\left(h\left[\frac{\alpha_i}{\beta_i^{\xi_t}}+\beta_i^{1-\xi_t}\,\tilde g_{M,t}\right]\right)}
  \]
  Here $\tilde g_{i,t}=\alpha_i+\beta_i\tilde g_{M,t}$ is the SIM's expected growth today,
  $h=1$ year makes the argument dimensionless, and the model assumes $\beta_i>0$.
  \begin{itemize}\setlength{\itemsep}{0.2em}
    \item \hilite{Sign picks the basket}: $\gamma_i(t)>0$ when $\tilde g_{M,t}>-\alpha_i/\beta_i$.
      As $\tilde g_{M,t}$ rises, the basket grows. As it falls, only the
      lowest-threshold assets stay. With an empty basket, the allocator holds the floors plus cash.
    \item \hilite{Magnitude splits the budget}: bearish ($\xi_t>0$) tilts it toward
      low-beta assets, bullish toward high-beta.
  \end{itemize}
\end{frame}

\begin{frame}{Market Inputs from SPY}
  \setlength{\abovedisplayskip}{2pt}\setlength{\belowdisplayskip}{1pt}
  Both inputs come from SPY prices through \hilite{exponential moving averages}
  (EMAs), which weight recent prices most. For windows of $L$ trading days:
  \[
    \bar S_t=\omega\,S_t+(1-\omega)\,\bar S_{t-1},\quad\omega=\frac{2}{L+1}
    \qquad\qquad
    \boxed{\xi_t=-G\left(\frac{\bar S^{\mathrm{short}}_t}{\bar S^{\mathrm{long}}_t}-1\right)}
  \]
  With $L_{\mathrm{short}}=21$ and $L_{\mathrm{long}}=63$, $\xi_t$ compares recent prices
  with the quarter's average:
  \begin{itemize}\setlength{\itemsep}{0em}
    \item $\bar S^{\mathrm{short}}_t>\bar S^{\mathrm{long}}_t$: $\xi_t<0$,
      \hilite{bullish}. The market has been rising.
    \item $\bar S^{\mathrm{short}}_t=\bar S^{\mathrm{long}}_t$: $\xi_t=0$,
      \hilite{neutral}, no tilt.
    \item $\bar S^{\mathrm{short}}_t<\bar S^{\mathrm{long}}_t$: $\xi_t>0$,
      \hilite{bearish}. The market has been falling.
  \end{itemize}
  We choose the gain $G>0$ so that $|\xi_t|\le1$ on 99\% of training days, and
  $\tilde g_{M,t}$ is the 252-day EMA of $g_{M,t}=\ln(S_t/S_{t-1})/\Delta t$, with $\Delta t=1/252$ year. Both
  inputs use prices through day $t-1$.

\end{frame}

% ==== mean-variance utility on the CAL ======================
\begin{frame}{The Complete Portfolio}
  As in L7a, a \hilite{complete portfolio} $c$ puts a fraction $w_f$ of wealth in the risk-free
  asset and $1-w_f$ in the tangent portfolio $\Tan$. Its expected growth rate and
  variance are:
  \[
    \begin{aligned}
      \E[g_c]&=g_f+(1-w_f)\bigl(\E[g_{\Tan}]-g_f\bigr) &&\quad\text{expectation is linear}\\
      \sigma^2_{g,c}&=(1-w_f)^2\,\sigma^2_{g,\Tan} &&\quad\text{the risk-free asset adds no variance}
    \end{aligned}
  \]
  where growth rates are in inverse years and $g_f$ is the risk-free growth rate.
  A negative $w_f$ means borrowing at $g_f$ to buy more of $\Tan$.

  \vspace{0.8em}
  \textbf{\ink{Which $w_f$ should a given investor choose?}} The answer depends on how
  risk averse the investor is.
\end{frame}

\begin{frame}{Measuring Risk Aversion}
  \setlength{\jot}{0pt}
  \setlength{\abovedisplayskip}{2pt}\setlength{\belowdisplayskip}{1pt}
  The \hilite{certainty equivalent} (CE) is the sure wealth whose utility equals the
  expected utility of an uncertain wealth $W$. The \hilite{risk premium} is the gap:
  \[
    \boxed{\underbrace{U(\CE)}_{\text{utility of the sure wealth}}\quad=\quad\underbrace{\E[U(W)]}_{\text{expected utility}}
    \qquad\qquad
    \RP=\underbrace{\E[W]}_{\text{expected wealth}}-\underbrace{\CE}_{\text{sure wealth}}}
  \]
  The more concave $U$, the further CE falls below $\E[W]$. Arrow and
  \href{https://doi.org/10.2307/1913738}{Pratt (1964)} measure this curvature with the
  \hilite{relative risk aversion} $\bar r$. Over one trading day, today's wealth
  $W_0$ grows to $W=W_0e^{g_c\Delta t}$:
  {\small
  \[
    \begin{aligned}
      \bar r&=-W_0\frac{U''(W_0)}{U'(W_0)} &&\quad\text{relative risk aversion}\\
      \RP&\approx-\frac12\,\frac{U''(W_0)}{U'(W_0)}\,\Var(W) &&\quad\text{Pratt (1964), small risk}\\
      \Var(W)&\approx W_0^2\,(\Delta t)^2\sigma^2_{g,c} &&\quad W\approx W_0(1+g_c\Delta t)\text{ for one day}\\
      \frac{\RP}{W_0}&\approx\frac{\bar r}{2}\,(\Delta t)^2\sigma^2_{g,c} &&\quad\text{substitute, divide by }W_0
    \end{aligned}
  \]
  }
\end{frame}

\begin{frame}{The Certainty-Equivalent Growth Rate}
  \setlength{\jot}{2pt}
  The \hilite{certainty-equivalent growth rate} $g_{\CE}$ is the sure rate that
  replaces the risky $g_c$:
  \[
    \begin{aligned}
      \CE&=W_0\,e^{g_{\CE}\Delta t} &&\text{definition of }g_{\CE}\\
      \frac{\E[W]}{W_0}&\approx1+\E[g_c]\,\Delta t+\tfrac12(\Delta t)^2\sigma^2_{g,c} &&\text{expand }e^{x}\text{, take expectations, }\E[g_c]^2\ll\sigma^2_{g,c}\\
      \frac{\CE}{W_0}&=\frac{\E[W]}{W_0}-\frac{\RP}{W_0} &&\RP=\E[W]-\CE\\
      &\approx1+\E[g_c]\,\Delta t-\frac{\bar r-1}{2}\,(\Delta t)^2\sigma^2_{g,c} &&\text{substitute, collect}\\
      g_{\CE}&\approx\E[g_c]-\frac{\bar r-1}{2}\,\Delta t\,\sigma^2_{g,c} &&\ln(1+x)\approx x\text{, divide by }\Delta t
    \end{aligned}
  \]
  The variance term lowers $g_{\CE}$ only when $\bar r>1$, which we assume from here on.
  \slidenote{Every step:}{\href{\repo/CHEME-5660-L8b-Lecture-Utility-Allocation-Bets-Fall-2026.ipynb}{lecture notebook}, Measuring risk aversion.}
\end{frame}

\begin{frame}{Mean-Variance Utility}
  The investor wants the largest sure growth rate. With $A\equiv\bar r-1$, $g_{\CE}$
  is the \hilite{mean-variance utility}, which we maximize over $w_f$:
  \[
    \boxed{\underset{w_f}{\operatorname{maximize}}\quad U(w_f)=\E[g_c]-\frac{A}{2}\,\Delta t\,\sigma^2_{g,c}\approx g_{\CE}}
  \]
  The risk-aversion coefficient $A>0$ (dimensionless) sets how much the investor
  dislikes variance. We assume $\E[g_{\Tan}]>g_f$. On the line, both terms depend
  only on $w_f$:
  \[
    \begin{aligned}
      U(w_f)&=g_f+(1-w_f)\bigl(\E[g_{\Tan}]-g_f\bigr)-\tfrac{A}{2}(1-w_f)^2\,\Delta t\,\sigma^2_{g,\Tan}
        &&\quad\text{substitute the line}\\
      \frac{dU}{dw_f}&=-\bigl(\E[g_{\Tan}]-g_f\bigr)+A\,(1-w_f)\,\Delta t\,\sigma^2_{g,\Tan}
        &&\quad\text{differentiate}\\
      0&=-\bigl(\E[g_{\Tan}]-g_f\bigr)+A\,(1-w_f^\star)\,\Delta t\,\sigma^2_{g,\Tan}
        &&\quad\text{first-order condition}
    \end{aligned}
  \]
\end{frame}

\begin{frame}{The Optimal Complete Portfolio}
  \setlength{\abovedisplayskip}{4pt}\setlength{\belowdisplayskip}{4pt}
  Solving the first-order condition for the fraction in $\Tan$ gives:
  \[
    \boxed{1-w_f^\star=\frac{\E[g_{\Tan}]-g_f}{A\,\Delta t\,\sigma^2_{g,\Tan}}}
  \]
  The second derivative, $-A\,\Delta t\,\sigma^2_{g,\Tan}$, is negative, so this is a
  maximum. The fraction in $\Tan$ rises with the excess growth and falls as $A$ or
  $\sigma^2_{g,\Tan}$ grows.

  \vspace{0.6em}
  An \hilite{indifference curve}, $\E[g_c]=U+\tfrac{A}{2}\,\Delta t\,\sigma^2_{g,c}$, joins the
  portfolios with the same utility $U$. The optimum is where the highest reachable
  curve touches the line.

  \vspace{0.6em}
  Solving the box for $A$ gives the risk aversion behind a client's chosen $w_f<1$.

\end{frame}


% ==== derivatives: a bet on the future =======================
\begin{frame}{Derivatives: A Bet on the Future}
  Suppose we pay $p$ today for a contract that pays 1 USD in $T$ years if
  $S_T\ge K$ (the L5a trade rule), and nothing otherwise.
  \begin{center}
    \includegraphics[width=\textwidth,height=0.40\textheight,keepaspectratio]{../figs/Fig-L8b-Bet-CashFlows.pdf}
  \end{center}
  Discounting the payoff at the benchmark growth rate $g_y$ gives a random NPV:
  \[
    \NPV_0=-p+\Dinv\,V(S_T)=\begin{cases}-p+e^{-g_yT}&\text{if }S_T\ge K\text{ (true)}\\-p&\text{if }S_T<K\text{ (false)}\end{cases}
  \]
\end{frame}

\begin{frame}{Expected NPV of the Bet}
  Under our beliefs about $S_T$, the expected NPV is:
  {\small
  \[
    \begin{aligned}
      \E[\NPV_0]&=\left(-p+e^{-g_yT}\right)\underbrace{\mathbb{P}(S_T\ge K)}_{\text{true}}+(-p)\,\underbrace{\mathbb{P}(S_T<K)}_{\text{false}} &&\text{weight by probability}\\
      &=-p\,\underbrace{\left[\mathbb{P}(S_T\ge K)+\mathbb{P}(S_T<K)\right]}_{=1}+e^{-g_yT}\,\mathbb{P}(S_T\ge K) &&\text{collect the }p\text{ terms}\\
      &=-p+e^{-g_yT}\,\underbrace{\mathbb{P}(S_T\ge K)}_{\text{true}} &&\text{probabilities sum to one}
    \end{aligned}
  \]
  }%
  The bet has a non-negative expected NPV when $p\le e^{-g_yT}\,\mathbb{P}(S_T\ge K)$.

  \vspace{0.8em}
  The bet is a \hilite{derivative contract}: its payoff depends on the future price
  of another asset. Every buyer pays one market price, yet buyers with different
  beliefs value the same contract differently.

  \vspace{0.8em}
  \textbf{\ink{How is the market price set?}}
\end{frame}

\begin{frame}{Summary}
\begin{itemize}\setlength{\itemsep}{0.9em}
\item \hilite{The single index model and Cobb--Douglas utility turn preferences into
holdings}: the signs of the preference weights pick the basket, and Cobb--Douglas
splits the net budget in proportion to the positive weights. The CES elasticity moves
the split toward concentration or equal share counts.
\item \hilite{Risk aversion picks the point on the capital allocation line}: the
certainty-equivalent growth rate is a mean-variance utility with $A=\bar r-1$, and
the risky fraction follows in closed form. A client's chosen fraction implies $A$.
\item \hilite{A bet on the future is an abstract asset with a random cash flow}: its
expected NPV depends on our beliefs, so buyers with different beliefs value it differently.
\end{itemize}
\end{frame}
