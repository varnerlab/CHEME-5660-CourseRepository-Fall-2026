\begin{frame}{What Is a Derivative?}
  A \hilite{derivative} is a contract between a buyer and a seller to transact an
  asset at a set price on a future date. In L1b's language, it is an abstract
  asset whose cash flows depend on the underlying asset.
  \vspace{0.3em}
  \begin{itemize}\setlength{\itemsep}{0.25em}
    \item \hilite{Futures and forwards}: agreements to buy or sell later at a set
      price. Futures are exchange-traded, and forwards trade over the counter.
    \item \hilite{Swaps}: one party exchanges an asset's value or cash flows for
      another's, over the counter.
    \item \hilite{Options}: the buyer holds the right, but not the obligation, to
      buy (call) or sell (put) at a set price. Options are exchange-traded.
    \item \hilite{Event contracts}: pay a fixed amount, usually one dollar, if an
      event occurs and nothing otherwise. Kalshi lists them.
  \end{itemize}
\end{frame}

  large but bounded, because the share price cannot fall below zero. Covered
  calls and cash-secured puts hold the shares or cash for 100 shares per

\begin{frame}{Call and Put Payoffs}
  A \hilite{call} gives the owner the right, but not the obligation, to buy 100
  shares at the strike price $K$ by the expiration date. A \hilite{put} gives
  the right to sell them. With share price $\ST$ at expiration, the payoffs per
  share are:
  \begin{equation*}
    \boxed{
    \begin{aligned}
      V_{c}(K,\ST) & = \max\bigl(\ST-K,~0\bigr)\\
      V_{p}(K,\ST) & = \max\bigl(K-\ST,~0\bigr)
    \end{aligned}}
  \end{equation*}
  The call is exercised only when $\ST>K$, and the put only when $\ST<K$. Both
  payoffs are nonnegative because the buyer may decline. The seller charges a
  premium, $\Pc$ for a call and $\Pp$ for a put.
\end{frame}

\begin{frame}{Profit and Breakeven}
  The buyer's profit per share is the payoff minus the premium, and the
  seller's is the premium minus the payoff:
  \begin{align*}
    P_{c}^{\text{buyer}} & = V_{c}(K,\ST)-\Pc & P_{c}^{\text{seller}} & = \Pc-V_{c}(K,\ST)\\
    P_{p}^{\text{buyer}} & = V_{p}(K,\ST)-\Pp & P_{p}^{\text{seller}} & = \Pp-V_{p}(K,\ST)
  \end{align*}
  Setting the buyer's profit to zero gives the breakeven share price at
  expiration:
  \[
    S_{\text{BE}}=K+\Pc\ \text{(call)},
    \qquad
    S_{\text{BE}}=K-\Pp\ \text{(put)}
  \]
  Above the call breakeven, and below the put breakeven, the buyer profits and
  the seller loses.

\end{frame}

\begin{frame}{Options as Abstract Assets}
  \vspace{-0.6em}
  \begin{center}
    \includegraphics[width=0.62\textwidth,keepaspectratio]{../figs/Fig-L9a-Option-CashFlows.pdf}
  \end{center}
  \vspace{-0.5em}
  The buyer pays the premium $\mathcal{P}$ today and receives the payoff
  $V(K,\ST)\geq 0$ at expiration, so the net present value at $t=0$ is:
  \[
    \operatorname{NPV}_{0}=-\mathcal{P}+\Dinv\cdot V(K,\ST)
  \]
  The profit $V-\mathcal{P}$ adds dollars from different dates. In today's
  dollars, the call buyer breaks even at $K+\Pc\cdot\mathcal{D}_{T,0}(g_{y})$
  rather than $K+\Pc$.
\end{frame}

\begin{frame}{The European Premium}
  A Treasury's cash flows are known, so in L2a its price made the NPV zero. An
  option's payoff is random, so we require a zero \hilite{expected} NPV under
  the risk-neutral measure $\Q$:
  \[
    \E_{\Q}\bigl(\operatorname{NPV}_{0}\bigr)=0
    \;\Longrightarrow\;
    \boxed{\mathcal{P}^{\text{EU}}=\E_{\Q}\bigl(\Dinv\cdot V(K,\ST)\bigr)}
  \]
  Any other premium creates a risk-free profit opportunity:
  \begin{itemize}\setlength{\itemsep}{0.3em}
    \item \hilite{Too low}: buy the option and sell a portfolio with the same
      payoff, built from shares and the risk-free asset.
    \item \hilite{Too high}: sell the option and buy the portfolio.
  \end{itemize}
\end{frame}

\begin{frame}{Exercise Styles}
  The exercise style changes the premium:
  \begin{itemize}\setlength{\itemsep}{0.25em}
    \item \hilite{European}: exercise only on the expiration date $T$.
    \item \hilite{American}: exercise at any time up to $T$.
    \item \hilite{Bermudan}: exercise only on specified dates up to $T$.
  \end{itemize}
  The American holder exercises at a time $\tau\leq T$ chosen using only the
  information available at that time (a stopping time):
  \[
    \mathcal{P}^{\text{AM}}=\sup_{\tau\leq T}\E_{\Q}\Bigl(\mathcal{D}^{-1}_{\tau,0}(g_{y})\cdot V(K,S(\tau))\Bigr)\geq\mathcal{P}^{\text{EU}}
  \]
  The American premium is at least the European one because the holder can
  choose when to exercise.
\end{frame}

\begin{frame}{What \texorpdfstring{$\Q$}{Q} Means}
  Under $\Q$, the expected share price grows at the risk-free rate $g_{y}$ (no
  dividends), so the share also has zero expected NPV:
  \[
    \E_{\Q}\bigl(\Dinv\cdot\ST\bigr)=S(0)
  \]
  $\Q$ is a pricing rule. The real-world measure $\mathbb{P}$ still describes
  how often outcomes occur.
  \begin{itemize}\setlength{\itemsep}{0.35em}
    \item \hilite{May have no value}: an option that expires worthless has
      scaled NPV $\rho_{T}=-1$. For a call, that happens with real-world
      probability $\mathbb{P}(\ST\leq K)$ (L5a).
    \item \hilite{Event contracts}: a one-dollar contract costs
      $p=\Dinv\cdot\Q(\text{event})$, a discounted $\Q$-probability that can
      differ from $\mathbb{P}(\text{event})$.
  \end{itemize}
\end{frame}

\begin{frame}{Black–Scholes–Merton Premiums}
  Under GBM with volatility $\sigma$, the European call and put premiums are:
  \begin{equation*}
    \boxed{
    \begin{aligned}
      \Pc & = N(d_{+})\cdot S(0)-N(d_{-})\cdot K\cdot\Dinv\\
      \Pp & = N(-d_{-})\cdot K\cdot\Dinv-N(-d_{+})\cdot S(0)
    \end{aligned}}
  \end{equation*}
  where $N$ is the standard normal CDF, $T$ is in years, and:
  \[
    d_{+}=\frac{1}{\sigma\sqrt{T}}\left[\ln\frac{S(0)}{K}+\Bigl(g_{y}+\frac{\sigma^{2}}{2}\Bigr)T\right],
    \qquad d_{-}=d_{+}-\sigma\sqrt{T}
  \]
  Merton and Scholes received the 1997 Nobel Prize in Economic Sciences.
  Fischer Black had passed away before the prize was awarded.
\end{frame}

\begin{frame}{Put–Call Parity}
  For a call and a put with the same strike and expiration, the payoffs differ
  by $\ST-K$ at every share price:
  \[
    V_{c}(K,\ST)-V_{p}(K,\ST)=\ST-K
  \]
  Subtracting the put premium condition from the call premium condition, and
  using $\E_{\Q}\bigl(\Dinv\cdot\ST\bigr)=S(0)$, gives:
  \[
    \boxed{\Pc-\Pp=S(0)-K\cdot\Dinv}
  \]
  Parity uses only the payoff identity and the share's zero expected NPV, so it
  holds whatever model describes the share price under $\Q$.

\end{frame}
