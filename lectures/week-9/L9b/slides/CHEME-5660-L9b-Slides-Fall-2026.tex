\begin{frame}{Concept Review: The European Premium}
  \vspace{0.3em}
  {\large\textbf{\ink{How do we compute the premium of a European option?}}}

  \vspace{0.5em}
  \begin{itemize}\setlength{\itemsep}{0.4em}
    \item \hilite{Previously}: we set the contract's expected NPV to zero under
      the risk-neutral pricing rule $\Q$.
    \item \hilite{Under $\Q$}: the expected share price grows at the risk-free
      rate $g_{y}$.
  \end{itemize}

  \vspace{0.5em}
  The premium is the discounted expected payoff under $\Q$:
  \[
    \boxed{\PEU_{\star}=\Dinv\cdot\E_{\Q}\bigl[V_{\star}(K,\ST)\bigr]\quad\text{where }\star\in\{c,p\}}
  \]
  where $V_{\star}(K,\ST)$ is the call or put payoff, $K$ is the strike price,
  $\ST$ is the share price at expiration $T$ (years), and $\Dinv=e^{-g_{y}T}$.
\end{frame}

\begin{frame}{Concept Review: A Monte Carlo Premium}
  Simulation estimates the market premium directly. The payoff depends only on
  $\ST$, so we draw it from the solution of geometric Brownian motion (GBM) with
  drift $g_{y}$ and volatility $\sigma$.

  \vspace{0.5em}
  \hilite{Initialize}: given $S(0)$, $K$, the payoff $V(K,S)$, $T$ (years),
  $\sigma$, $g_{y}$, and $M$ paths.
  \begin{enumerate}\setlength{\itemsep}{0.15em}
    \item For each path $i=1,2,\dots,M$ \textbf{do}:
      \begin{itemize}
        \item[a.] Draw a standard normal random number $Z_{i}$.
        \item[b.] $S_{i}(T)\gets S(0)\cdot\exp\bigl[(g_{y}-\sigma^{2}/2)\,T+\sigma\sqrt{T}\,Z_{i}\bigr]$.
        \item[c.] $V_{i}\gets V(K,S_{i}(T))$.
      \end{itemize}
    \item Discount the average payoff:
      $\mathcal{P}\gets\Dinv\cdot\frac{1}{M}\sum_{i=1}^{M}V_{i}$.
  \end{enumerate}
  The result is an estimate: it changes with the random draws, and it becomes
  more precise as $M$ grows.
  \slidenote{Example (Approach 1 in Tasks 1 and 2):}{\href{\repoprev/CHEME-5660-L9a-Example-BSM-Premium-Fall-2026.ipynb}{\dimmed{CHEME-5660-L9a-Example-BSM-Premium-Fall-2026.ipynb}}\newline

\end{frame}

\begin{frame}{The Black–Scholes–Merton Equation}
  Merton and Scholes received the 1997 Nobel Prize in Economic Sciences for this
  model. Fischer Black had passed away before the award.

  \vspace{0.5em}
  Assume the price of a non-dividend-paying share follows GBM with constant
  $\sigma$ and a constant, continuously compounded risk-free rate $g_{y}$. The premium $\mathcal{P}(S,t)$, with
  $\mathcal{P}(S,T)=V(K,S)$, satisfies:
  \[
    \boxed{\frac{\partial\mathcal{P}}{\partial t}+g_{y}\,S\,\frac{\partial\mathcal{P}}{\partial S}+\frac{1}{2}\,\sigma^{2}S^{2}\,\frac{\partial^{2}\mathcal{P}}{\partial S^{2}}=g_{y}\,\mathcal{P}}
  \]
  Solving it backward in time from the payoff gives today's premium,
  $\mathcal{P}(S(0),0)$, and this solution equals the risk-neutral premium
  $\mathcal{P}^{\text{EU}}$ that the Monte Carlo simulation estimates.
  \slidenote{Derivation:}{the \href{\repo/advanced/contingent_claims/CHEME-5660-L9b-Advanced-ContingentClaims-BSM-Fall-2026.ipynb}{contingent claims notebook}.}
\end{frame}

\begin{frame}{Black–Scholes–Merton Premiums}
  Instead of simulating, we can compute this premium with the closed-form
  European call and put formulas:
  \begin{equation*}
    \boxed{
    \begin{aligned}
      \Pc & = N(d_{+})\cdot S(0)-N(d_{-})\cdot K\cdot\Dinv\\
      \Pp & = N(-d_{-})\cdot K\cdot\Dinv-N(-d_{+})\cdot S(0)
    \end{aligned}}
  \end{equation*}
  where $N$ is the standard normal CDF, $T$ is in years, and:
  \[
    d_{+}=\frac{1}{\sigma\sqrt{T}}\left[\ln\left(\frac{S(0)}{K}\right)+\Bigl(g_{y}+\frac{\sigma^{2}}{2}\Bigr)T\right],
    \qquad d_{-}=d_{+}-\sigma\sqrt{T}
  \]
  We know $S(0)$, $K$, $T$, and $g_{y}$ when we price the contract. The
  volatility $\sigma$ is the one input we must choose.
\end{frame}

\begin{frame}{Which Volatility Should We Use?}
  If we knew the right volatility, the BSM model (or the Monte Carlo model)
  would compute the observed market premium. In practice, we observe the market
  premium and back out the volatility that makes the model match it.
  \begin{itemize}\setlength{\itemsep}{0.6em}
    \item \hilite{Historical volatility}:
      $\sigma_{\mathrm{ann}}=\sqrt{\Delta t}\cdot\sigma_{g}$, where $\sigma_{g}$
      is the standard deviation of past growth rates (units: $\mathrm{yr}^{-1}$)
      and $\Delta t$ is the time between observations (units: years). It looks
      backward.
    \item \hilite{Implied volatility}
      (\href{https://en.wikipedia.org/wiki/Implied_volatility}{IV}): the
      market's forecast of a likely future movement in the share price. It looks
      forward, and it is often at or above the historical volatility.
    \item \hilite{How can we use it?} IV represents a one-standard-deviation
      movement in the share price over the next year. Over $T$ years, the move
      scales with $\sqrt{T}$.
  \end{itemize}
  \vspace{0.3em}
  This $\sqrt{T}$ scaling gives a quick range for the share price on the next
  slide.
\end{frame}

\begin{frame}{A Back-of-the-Envelope Range from IV}
  Moving one standard deviation up or down in the solution of geometric
  Brownian motion, $Z=\pm1$, gives a back-of-the-envelope range for the share
  price:
  {\small
  \begin{equation*}
    \begin{aligned}
      S(T) & = S(0)\exp\left[\mu_{g}T + \sigma\sqrt{T}\,Z\right] && \text{solution of geometric Brownian motion}\\
      S_{\pm} & = S(0)\exp\left[\mu_{g}T \pm \text{IV}\sqrt{T}\right] && \text{set } \sigma=\text{IV} \text{ and } Z=\pm 1\\
      & \approx S(0)\exp\left[\pm\,\text{IV}\sqrt{T}\right] && \mu_{g}T \text{ is small next to } \text{IV}\sqrt{T} \text{ for short } T\\
      & \approx S(0)\left(1 \pm \text{IV}\sqrt{T}\right) && e^{x}\approx 1+x \text{ for small } x
    \end{aligned}
  \end{equation*}}%
  where $\mu_{g}$ is the mean growth rate, $Z$ is standard normal, and $S_{\pm}$
  is the share price one standard deviation above or below $S(0)$. Under the
  model, the share price ends inside this range about 68\% of the time.
\end{frame}

\begin{frame}{Implied Volatility by Bisection}
  We back out the implied volatility $\hat{\sigma}$ by bisection.

  \vspace{0.3em}
  \hilite{Initialize}: given $\mathcal{P}^{\text{market}}$, $S(0)$, $K$, $T$
  (years), $g_{y}$, a bracket $\sigma_{\text{lo}}<\sigma_{\text{hi}}$ with
  $\mathcal{P}(\sigma_{\text{lo}})\leq\mathcal{P}^{\text{market}}\leq\mathcal{P}(\sigma_{\text{hi}})$,
  a tolerance $\epsilon>0$, $\texttt{maxiter}$, and $i\gets 1$.

  \vspace{0.3em}
  While $\sigma_{\text{hi}}-\sigma_{\text{lo}}>\epsilon$ and $i\leq\texttt{maxiter}$ \textbf{do}:
  \begin{enumerate}\setlength{\itemsep}{0.2em}
    \item $\sigma_{\text{mid}}\gets(\sigma_{\text{lo}}+\sigma_{\text{hi}})/2$
      and compute its BSM premium $\mathcal{P}(\sigma_{\text{mid}})$.
    \item If $\mathcal{P}(\sigma_{\text{mid}})<\mathcal{P}^{\text{market}}$,
      \textbf{then} $\sigma_{\text{lo}}\gets\sigma_{\text{mid}}$. Otherwise,
      $\sigma_{\text{hi}}\gets\sigma_{\text{mid}}$.
    \item $i\gets i+1$.
  \end{enumerate}
  The implied volatility is the midpoint of the final bracket,
  $\hat{\sigma}\gets(\sigma_{\text{lo}}+\sigma_{\text{hi}})/2$. The BSM premium
  rises with $\sigma$, so exactly one $\sigma$ matches the market premium, and
  each step halves the bracket around it.
\end{frame}

\begin{frame}{Implied Volatility in Practice}
  Three points about using implied volatility with real quotes:
  \begin{itemize}\setlength{\itemsep}{0.6em}
    \item \hilite{Market quotes}: the SPXW volatility skew notebook applies this
      bisection to a chain of S\&P 500 index options.
    \item \hilite{One per contract}: each strike and expiration has its own
      implied volatility, although the BSM model assumes one constant $\sigma$.
      This pattern is known as the volatility smile or volatility skew.
    \item \hilite{Pricing without a quote}: to price a contract without a quote,
      or on a future date, we can predict its implied volatility from a surface
      fitted to today's quotes, which gives implied volatility as a function of
      strike and expiration
      (\href{https://arxiv.org/abs/2605.13998}{Sun et al., 2026}).
  \end{itemize}

\begin{frame}{Interesting Fact: Put–Call Parity}
  For a European call and put with the same strike and expiration, the payoffs differ by:
  {\small\setlength{\jot}{1pt}%
  \setlength{\abovedisplayskip}{3pt}\setlength{\belowdisplayskip}{3pt}%
  \[
    V_{c}(K,\ST)-V_{p}(K,\ST)=\max\bigl(\ST-K,~0\bigr)-\max\bigl(K-\ST,~0\bigr)=\ST-K
  \]
  \vspace{-0.6em}
  \[
    \begin{alignedat}{2}
      \Pc-\Pp&=\Dinv\cdot\E_{\Q}\bigl[V_{c}(K,\ST)\bigr] &\enspace&\text{\footnotesize premium formula}\\
      &\qquad-\Dinv\cdot\E_{\Q}\bigl[V_{p}(K,\ST)\bigr] &&\\
      &=\Dinv\cdot\E_{\Q}\bigl[V_{c}(K,\ST)-V_{p}(K,\ST)\bigr] &&\text{\footnotesize expectation is linear}\\
      &=\Dinv\cdot\E_{\Q}\bigl[\ST-K\bigr] &&\text{\footnotesize payoff identity}\\
      &=\Dinv\cdot\bigl(S(0)\,e^{g_{y}T}-K\bigr) &&\text{\footnotesize grows at }g_{y}\text{\footnotesize\ under }\Q\\
      &=S(0)-K\cdot\Dinv &&\Dinv=e^{-g_{y}T}
    \end{alignedat}
  \]
  }%
  This is \hilite{put--call parity}: $\boxed{\Pc-\Pp=S(0)-K\cdot\Dinv}$\par
  It holds only for European contracts: an American holder can exercise early.
  \slidenote{Interesting fact:}{\href{\repo/CHEME-5660-L9b-InterestingFact-PutCallParity-Fall-2026.ipynb}{\dimmed{CHEME-5660-L9b-InterestingFact-PutCallParity-Fall-2026.ipynb}}}
\end{frame}

\begin{frame}{American Style Options Contracts}
  American-style contracts can be exercised \hilite{at any time} before or on
  the expiration date~$T$. We write the American premium as:
  \begin{equation*}
    \PAM_{\star}=\max_{\tau\leq T}\E_{\Q}\Bigl(\mathcal{D}^{-1}_{\tau,0}(g_{y})\cdot V_{\star}(K,S(\tau))\Bigr)\geq\PEU_{\star}\qquad\text{for }\star\in\{c,p\}
  \end{equation*}
  \begin{itemize}\setlength{\itemsep}{0.5em}
    \item \hilite{Stopping time}: $\tau$ uses only the information available at that time.
    \item \hilite{Discounting}: $\mathcal{D}^{-1}_{\tau,0}(g_{y})$ brings the payoff at exercise time $\tau$ back to today.
    \item \hilite{At least the European premium}: the holder can always wait.
  \end{itemize}
  \vspace{0.2em}
  The standard pricing method is the Cox, Ross, Rubinstein (CRR) binomial model.
  \slidenote{Reference:}{Cox, Ross, and Rubinstein (1979). Option pricing: A simplified approach. \href{https://www.sciencedirect.com/science/article/pii/0304405X79900151}{J. Financial Economics 7(3): 229}.}
\end{frame}

\begin{frame}{CRR Binomial Lattice Model}
  The share price moves up by $u$ with \hilite{risk-neutral probability} $q$, or down by $d$:
  \begin{equation*}
    \begin{aligned}
      \mathcal{D}_{1,0}(g_{y})\cdot S_{0} & = \E_{\Q}\left(S_{1}\right) && \text{grow at the risk-free rate } g_{y} \text{ under }\Q\\
      & = q\cdot(u\cdot S_{0}) + (1-q)\cdot(d\cdot S_{0}) && \text{up with probability } q\text{, down with } (1-q)\\
      & = S_{0}\bigl(qu + (1-q)d\bigr) && \text{factor out } S_{0}\\
      & = S_{0}\bigl(qu + d - qd\bigr) && \text{expand } (1-q)d\\
      & = S_{0}\bigl(d + q(u - d)\bigr) && \text{collect the } q \text{ terms}\\[0.3em]
      \Longrightarrow\quad & \boxed{q = \frac{\mathcal{D}_{1,0}(g_{y}) - d}{u - d}} && \text{divide by } S_{0}\text{ and solve for } q
    \end{aligned}
  \end{equation*}
  where $S_{0}=S(0)$ is today's share price and $R_{f}=\mathcal{D}_{1,0}(g_{y})=e^{g_{y}\Delta t}$ for one step $\Delta t$.
\end{frame}

\begin{frame}{CRR Up and Down Factors}
  With $ud=1$, let $a=\ln u=-\ln d$. Match the growth rate $g=\pm a/\Delta t$ to GBM:
  \begin{equation*}
    \begin{aligned}
      \frac{(2p-1)\,a}{\Delta t} & = \mu_{g} && \text{match the mean growth rate}\\
      \frac{4p(1-p)\,a^{2}}{\Delta t^{2}} & = \frac{\sigma^{2}}{\Delta t} && \text{match the variance}\\
      \frac{a^{2}}{\Delta t^{2}} - \mu_{g}^{2} & = \frac{\sigma^{2}}{\Delta t} && \text{use } 4p(1-p) = 1-(2p-1)^{2}\text{ and the mean}\\
      a^{2} & = \sigma^{2}\Delta t + \mu_{g}^{2}\Delta t^{2} && \text{solve for } a^{2}\\[0.3em]
      \Longrightarrow\quad & \boxed{u = e^{\sigma\sqrt{\Delta t}},\quad d = e^{-\sigma\sqrt{\Delta t}}} && \text{keep the first term, so } a=\sigma\sqrt{\Delta t}
    \end{aligned}
  \end{equation*}
  where $p$ is the real-world up probability, $\mu_{g}$ the mean growth rate, and $\sigma$ the volatility.
  \slidenote{Where do $u$ and $d$ come from?}{\href{\repo/advanced/crr_factors/CHEME-5660-L9b-Advanced-CRR-Factors-Fall-2026.ipynb}{\dimmed{advanced/crr\_factors/CHEME-5660-L9b-Advanced-CRR-Factors-Fall-2026.ipynb}}}
\end{frame}

\begin{frame}{The Early-Exercise Premium}
  Working backward from expiration, the American holder takes the larger of
  exercising and holding, while the European holder can only hold:
  {\small
  \begin{equation*}
    \boxed{
    \begin{aligned}
      \hat{V}^{\text{AM}}_{t,k} & = \max\Bigl\{\underbrace{V(K,S_{t,k})}_{\text{exercise now}},~\underbrace{\Dstep\cdot\overbrace{\bigl(q\,\hat{V}^{\text{AM}}_{t+1,k+1} + (1-q)\,\hat{V}^{\text{AM}}_{t+1,k}\bigr)}^{\textstyle\mathbb{E}_{\mathbb{Q}}}}_{\text{hold for the future}}\Bigr\}\\
      \hat{V}^{\text{EU}}_{t,k} & = \Dstep\cdot\overbrace{\bigl(q\,\hat{V}^{\text{EU}}_{t+1,k+1} + (1-q)\,\hat{V}^{\text{EU}}_{t+1,k}\bigr)}^{\textstyle\mathbb{E}_{\mathbb{Q}}}
    \end{aligned}}
  \end{equation*}}%
  where $\hat{V}_{t,k}$ is the contract's value with $k$ up moves in $t$ steps,
  $S_{t,k}=S_{0}u^{k}d^{t-k}$, and $\Dstep=e^{-g_{y}\Delta t}$. The premium is
  $\mathcal{P}=\hat{V}_{0,0}$.

  \vspace{0.5em}
  \hilite{Early-exercise premium}: $\PAM-\PEU\geq 0$.
\end{frame}

\begin{frame}{Pseudocode: Pricing on the CRR Lattice}
  \setbeamerfont{itemize/enumerate subsubbody}{size=\small}
  \hilite{Initialize}: given $S_{0}$, $K$, the payoff function $V(K,S)$ of the
  call or put, $T$, $\sigma$, $g_{y}$, and the number of time steps $h$. Compute
  $\Delta t=T/h$, $u$ and $d$, $q$, and $\Dstep=e^{-g_{y}\Delta t}$.
  \begin{enumerate}\setlength{\itemsep}{0.3em}
    \item Build the share price lattice: $S_{t,k}\gets S_{0}u^{k}d^{t-k}$ for
      each level $t=0,1,\dots,h$ and node $k=0,1,\dots,t$.
    \item At expiration, set each node value to the payoff:
      $\hat{V}_{h,k}\gets V(K,S_{h,k})$ for $k=0,1,\dots,h$.
    \item For each level $t=h-1,h-2,\dots,0$ \textbf{do}:
      \begin{itemize}
        \item For each node $k=0,1,\dots,t$ \textbf{do}:
          \begin{itemize}
            \item[a.] Compute the value of holding for the future:
              $H_{t,k}\gets\Dstep\cdot\bigl(q\,\hat{V}_{t+1,k+1}+(1-q)\,\hat{V}_{t+1,k}\bigr)$.
            \item[b.] If the contract is American, \textbf{then}:
              $\hat{V}_{t,k}\gets\max\bigl\{V(K,S_{t,k}),~H_{t,k}\bigr\}$.
              Otherwise, $\hat{V}_{t,k}\gets H_{t,k}$.
          \end{itemize}
      \end{itemize}
    \item The premium is the value at the root: $\mathcal{P}\gets\hat{V}_{0,0}$.
  \end{enumerate}
\end{frame}

\begin{frame}{A Hypothetical Example from Hull}
  Consider a put contract.
  \begin{itemize}\setlength{\itemsep}{0.4em}
    \item \hilite{Contract}: the strike price is $K=52$ USD/share on a stock
      whose current price is $S_{0}=50$ USD/share.
    \item \hilite{Lattice}: two time steps of 1 year each. In each period, the
      stock price moves up by a factor of $u=1.2$ or down by a factor of $d=0.8$.
    \item \hilite{Pricing inputs}: the continuously compounded risk-free rate is
      $g_{y}=5\%$, so the risk-neutral probability of an up move is
      $q=(e^{0.05}-0.8)/(1.2-0.8)=0.6282$.
  \end{itemize}

\begin{frame}{Summary}
  Today we priced European options with the BSM formulas, and American options
  with the CRR binomial model.

  \vspace{0.3em}
  {\large\textbf{\ink{Key takeaways}}}
  \begin{itemize}\setlength{\itemsep}{0.15em}
    \item \hilite{The Black--Scholes--Merton formulas price European options}:
      they match the Monte Carlo premium, and we backed out the implied
      volatility from a market premium by bisection.
    \item \hilite{Binomial trees price American options}: at each node, the
      holder exercises or holds, so backward induction computes the premium.
    \item \hilite{The decision rule sets the early-exercise premium}: in
      Hull's put, exercising at the one node where it beats holding accounts
      for the whole gap between the American and European premiums.
  \end{itemize}

  \vspace{0.1em}
  The early-exercise premium is what the right to exercise before expiration is
  worth, and the binomial lattice computes it.
\end{frame}
